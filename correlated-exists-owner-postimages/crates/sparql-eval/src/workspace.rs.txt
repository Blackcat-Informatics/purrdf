//! Execution-owned admission, shared by nested work and retained publication.

use std::sync::{Mutex, MutexGuard};

use purrdf_core::{OwnedWorkspaceReservation, small::Shared};

use crate::EvalError;

/// Shared execution controls keep their actual payload and admission together.
/// A caller-provided Arc remains caller-owned; native controls use the same
/// immutable, fallible Shared home as rows and parsed values.
#[derive(Debug)]
pub(crate) struct SharedWorkspace<T>(SharedWorkspaceStorage<T>);

#[derive(Debug)]
enum SharedWorkspaceStorage<T> {
    Caller(std::sync::Arc<T>),
    Native(Shared<SharedWorkspacePayload<T>>),
}

#[derive(Debug)]
struct SharedWorkspacePayload<T> {
    value: T,
    // The payload must die before its original grant.
    _allocation: WorkspaceAllocation,
}

impl<T> Clone for SharedWorkspace<T> {
    fn clone(&self) -> Self {
        Self(match &self.0 {
            SharedWorkspaceStorage::Caller(value) => SharedWorkspaceStorage::Caller(value.clone()),
            SharedWorkspaceStorage::Native(value) => SharedWorkspaceStorage::Native(value.clone()),
        })
    }
}

impl<T> std::ops::Deref for SharedWorkspace<T> {
    type Target = T;
    fn deref(&self) -> &T {
        match &self.0 {
            SharedWorkspaceStorage::Caller(value) => value,
            SharedWorkspaceStorage::Native(value) => &value.value,
        }
    }
}

impl<T> AsRef<T> for SharedWorkspace<T> {
    fn as_ref(&self)->&T { self }
}

impl<T> From<std::sync::Arc<T>> for SharedWorkspace<T> {
    fn from(value: std::sync::Arc<T>) -> Self {
        Self(SharedWorkspaceStorage::Caller(value))
    }
}

impl<T> SharedWorkspace<T> {
    pub(crate) fn unique_mut(&mut self) -> Option<&mut T> {
        match &mut self.0 {
            SharedWorkspaceStorage::Caller(value) => std::sync::Arc::get_mut(value),
            SharedWorkspaceStorage::Native(value) => value.get_mut().map(|payload| &mut payload.value),
        }
    }

    /// Only explicitly caller-owned resident controls may be mutated. A native
    /// shared owner never exposes mutable payload ownership or its admission.
    pub(crate) fn caller_unique_mut(&mut self) -> Option<&mut T> {
        match &mut self.0 {
            SharedWorkspaceStorage::Caller(value) => std::sync::Arc::get_mut(value),
            SharedWorkspaceStorage::Native(_) => None,
        }
    }

    /// Borrow the resident public handle without copying its payload. Native
    /// controls deliberately expose no Arc conversion or unadmitted copy.
    pub(crate) fn caller_arc(&self) -> Option<std::sync::Arc<T>> {
        match &self.0 {
            SharedWorkspaceStorage::Caller(value) => Some(value.clone()),
            SharedWorkspaceStorage::Native(_) => None,
        }
    }

    pub(crate) fn control_bytes() -> usize {
        Shared::<SharedWorkspacePayload<T>>::allocation_layout().size()
    }

    pub(crate) fn new_admitted(
        value: T,
        workspace: &WorkspaceCapability,
    ) -> Result<Self, EvalError> {
        let bytes = fallible_shared_layout::<SharedWorkspacePayload<T>>()?;
        let allocation = workspace.charge(bytes)?;
        let payload = Shared::try_new(SharedWorkspacePayload {
            value,
            _allocation: allocation,
        })
        .map_err(|_| EvalError::AllocationFailed {
            construct: "shared execution control",
        })?;
        Ok(Self(SharedWorkspaceStorage::Native(payload)))
    }
}

/// One original grant for native lexical buffers and controls. Native Memory
/// admits every replacement while old buffers remain alive; the frame stores
/// the first typed evaluator refusal independently of its static leaf marker.
#[derive(Debug)]
pub(crate) struct LexicalFrame {
    workspace: WorkspaceCapability,
    allocation: Option<WorkspaceAllocation>,
    live: usize,
    failure: Option<EvalError>,
}

impl LexicalFrame {
    pub(crate) fn new(workspace: &WorkspaceCapability) -> Self {
        Self {
            workspace: workspace.clone(),
            allocation: None,
            live: 0,
            failure: None,
        }
    }

    /// Adopt the original private query grant without allocating or resizing it.
    /// `allocation` must belong to `workspace` and cover the exact requested
    /// layout total `live_bytes` of the still-live query payload. Native callers
    /// resume Memory with that same total; they never reset a live AST to zero.
    pub(crate) fn from_allocation(
        workspace: &WorkspaceCapability,
        allocation: WorkspaceAllocation,
        live_bytes: usize,
    ) -> Self {
        Self {
            workspace: workspace.clone(),
            allocation: Some(allocation),
            live: live_bytes,
            failure: None,
        }
    }

    pub(crate) const fn live_bytes(&self) -> usize {
        self.live
    }
    pub(crate) const fn workspace(&self) -> &WorkspaceCapability {
        &self.workspace
    }
    pub(crate) fn take_failure(&mut self) -> Option<EvalError> {
        self.failure.take()
    }

    pub(crate) fn resize_live(&mut self, live: usize) -> Result<(), EvalError> {
        if self.failure.is_some() {
            return Err(EvalError::WorkspaceStopped);
        }
        let resized: Result<(), EvalError> = (|| {
            let bytes = u64::try_from(live).map_err(|_| EvalError::WorkspaceBoundOverflow)?;
            match &mut self.allocation {
                Some(allocation) => allocation.resize(bytes)?,
                None if live != 0 => self.allocation = Some(self.workspace.charge(bytes)?),
                None => {}
            }
            self.live = live;
            Ok(())
        })();
        if let Err(error) = &resized {
            // These resize failures are fixed-layout/static control errors.
            // The provider's original typed cause remains in QueryWorkspace.
            self.failure = Some(error.clone());
        }
        resized
    }

    pub(crate) fn storage_error(
        &mut self,
        error: purrdf_lex::allocation::StorageError,
        construct: &'static str,
    ) -> EvalError {
        self.take_failure().unwrap_or(match error {
            purrdf_lex::allocation::StorageError::SizeOverflow => EvalError::WorkspaceBoundOverflow,
            purrdf_lex::allocation::StorageError::AllocationFailed => {
                EvalError::AllocationFailed { construct }
            }
            purrdf_lex::allocation::StorageError::AdmissionFailed => EvalError::WorkspaceStopped,
            purrdf_lex::allocation::StorageError::FormattingFailed => {
                EvalError::UnstableNativeDiagnostic
            }
        })
    }

    /// Preserve a native diagnostic's original payload admission at publication.
    /// Refusal leaves its text and original grant enclosed by the caller.
    pub(crate) fn finish_diagnostic(
        &mut self,
        diagnostic: &mut Option<purrdf_core::RdfDiagnostic>,
    ) -> Result<crate::RetainedDiagnostic, EvalError> {
        if let Some(error) = self.take_failure() {
            return Err(error);
        }
        let total = self.live.checked_add(crate::RetainedDiagnostic::control_bytes())
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        self.resize_live(total)?;
        let value = diagnostic.take().expect("native diagnostic publication owns its payload");
        let allocation = self.allocation.take().expect("diagnostic control has an original grant");
        self.live = 0;
        crate::RetainedDiagnostic::from_admitted(value, allocation)
    }

    /// Trusted native publications move their buffers without allocating.
    /// Every refusal is selected before the factory materializes the value.
    pub(crate) fn finish_term(
        &mut self,
        factory: impl FnOnce() -> purrdf_core::TermValue,
    ) -> Result<WorkspaceTerm, EvalError> {
        if let Some(error) = self.take_failure() {
            return Err(error);
        }
        if self.allocation.is_none() {
            self.allocation = Some(self.workspace.charge(0)?);
        }
        let allocation = self
            .allocation
            .take()
            .expect("original native grant checked before publication");
        let value = factory();
        Ok(WorkspaceTerm::new(value, allocation))
    }

    /// Publish one originally admitted UTF-8 buffer without copying its bytes.
    /// Construction scratch must already be destroyed, leaving exactly this
    /// buffer in the frame. Refusal leaves the enclosing text and grant intact.
    pub(crate) fn finish_text(
        &mut self,
        text: &mut Option<String>,
    ) -> Result<purrdf_lex::allocation::SharedText, EvalError> {
        use crate::error::{NativeDiagnostic, NativeDiagnosticKind};
        if let Some(error) = self.take_failure() {
            return Err(error);
        }
        let Some(capacity) = text.as_ref().map(String::capacity) else {
            return Err(NativeDiagnostic::error(
                NativeDiagnosticKind::Internal,
                &"native text publication has no payload",
                &self.workspace,
            ));
        };
        if capacity != self.live {
            return Err(NativeDiagnostic::error(
                NativeDiagnosticKind::Internal,
                &"native text publication retains unmatched working storage",
                &self.workspace,
            ));
        }
        let control =
            purrdf_lex::allocation::SharedText::allocation_layout::<WorkspaceAllocation>().size();
        let bytes = capacity
            .checked_add(control)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        self.resize_live(bytes)?;
        // Every fallible admission occurs while both original owners remain
        // enclosing. The native SharedText factory keeps text before its grant
        // even when the actual control allocator refuses.
        let allocation = self
            .allocation
            .take()
            .expect("positive shared control has its original grant");
        let text = text
            .take()
            .expect("native text checked before ownership transfer");
        self.live = 0;
        purrdf_lex::allocation::SharedText::try_from_admitted(text, allocation).map_err(|_| {
            let error = EvalError::AllocationFailed {
                construct: "native text shared control",
            };
            self.failure = Some(error.clone());
            error
        })
    }

    /// Transfer the native surviving layout's original grant with its payload.
    /// No raw public buffer or mutable admitted owner is exposed.
    pub(crate) fn finish_dataset(
        &mut self,
        dataset: &mut Option<purrdf_core::RdfDataset>,
    ) -> Result<purrdf_core::DatasetHandle, EvalError> {
        if let Some(error) = self.take_failure() {
            return Err(error);
        }
        let control = purrdf_core::DatasetHandle::allocation_layout::<WorkspaceAllocation>().size();
        let bytes = self
            .live
            .checked_add(control)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        self.resize_live(bytes)?;
        let value = dataset
            .take()
            .expect("native frozen dataset is present before publication");
        let allocation = self
            .allocation
            .take()
            .expect("native dataset control has original admission");
        self.live = 0;
        purrdf_core::DatasetHandle::try_from_admitted(value, allocation).map_err(|_| {
            EvalError::AllocationFailed {
                construct: "native dataset shared control",
            }
        })
    }

    /// Transfer the native surviving layout's original grant with its payload.
    /// No raw public buffer or mutable admitted owner is exposed.
    pub(crate) fn into_allocation(self) -> Option<WorkspaceAllocation> {
        self.allocation
    }
}

impl purrdf_lex::allocation::Admission for LexicalFrame {
    fn resize(&mut self, live: usize) -> Result<(), purrdf_lex::allocation::StorageError> {
        self.resize_live(live)
            .map_err(|_| purrdf_lex::allocation::StorageError::AdmissionFailed)
    }
}

impl purrdf_sparql_algebra::parser::ParserAdmission for LexicalFrame {
    fn text(
        &mut self,
        text: &dyn core::fmt::Display,
    ) -> Result<purrdf_lex::allocation::SharedText, purrdf_lex::allocation::StorageError> {
        if self.failure.is_some() {
            return Err(purrdf_lex::allocation::StorageError::AdmissionFailed);
        }
        self.workspace.authored_text(text).map_err(|error| {
            self.failure = Some(error);
            purrdf_lex::allocation::StorageError::AdmissionFailed
        })
    }
}

trait GrowthAdmission: std::fmt::Debug + Send + Sync {
    fn has_failed(&self) -> bool;
    fn charge(&self, bytes: u64) -> Result<WorkspaceAllocation, EvalError>;
    fn retain(&self, bytes: u64) -> Result<(), EvalError>;
}

trait AllocationOwner: std::fmt::Debug + Send + Sync {
    fn resize(&mut self, bytes: u64) -> Result<(), EvalError>;
}

/// Immutable report records retain the actual arrays and each record's native
/// payload grant. Moving records into a receipt never sheds those owners.
#[derive(Debug)]
pub(crate) struct AdmittedRecords<T> {
    values: AdmittedVec<T>,
    // Values and their buffers die before either payload or owner-array grants.
    allocations: AdmittedVec<WorkspaceAllocation>,
}

impl<T> std::ops::Deref for AdmittedRecords<T> {
    type Target = [T];
    fn deref(&self) -> &[T] {
        &self.values
    }
}

impl<T> AdmittedRecords<T> {
    pub(crate) fn with_capacity(
        capacity: usize,
        workspace: &WorkspaceCapability,
    ) -> Result<Self, EvalError> {
        Ok(Self {
            values: AdmittedVec::with_capacity(capacity, workspace)?,
            allocations: if workspace.is_bounded() {
                AdmittedVec::with_capacity(capacity, workspace)?
            } else {
                AdmittedVec::new(workspace)
            },
        })
    }

    pub(crate) fn new(workspace: &WorkspaceCapability) -> Self {
        Self {
            values: AdmittedVec::new(workspace),
            allocations: AdmittedVec::new(workspace),
        }
    }

    pub(crate) fn resident(values: Vec<T>) -> Self {
        let workspace = WorkspaceCapability::resident();
        Self {
            values: AdmittedVec::from_parts(values, None, &workspace),
            allocations: AdmittedVec::new(&workspace),
        }
    }

    pub(crate) fn push(
        &mut self,
        value: T,
        allocation: WorkspaceAllocation,
    ) -> Result<(), EvalError> {
        // The tuple drops its payload before its original grant on either refusal.
        let record = (value, allocation);
        self.values.reserve_next()?;
        if record.1.is_bounded() {
            self.allocations.reserve_next()?;
        }
        let (value, allocation) = record;
        self.values.push_reserved(value);
        if allocation.is_bounded() {
            self.allocations.push_reserved(allocation);
        }
        Ok(())
    }

    pub(crate) fn try_into_resident(self) -> Result<Vec<T>, Self> {
        if self.values.workspace.is_bounded() {
            return Err(self);
        }
        Ok(self.values.into_parts().0)
    }

    pub(crate) fn as_mut_slice(&mut self) -> &mut [T] {
        self.values.as_mut_slice()
    }
}

/// An internal vector whose immutable borrowed view cannot shed its owner.
/// Growth constructs a replacement while the original buffer stays covered,
/// so an allocator failure leaves the original allocation and grant intact.
#[derive(Debug)]
pub struct AdmittedVec<T> {
    values: Vec<T>,
    allocation: Option<WorkspaceAllocation>,
    workspace: WorkspaceCapability,
}

impl<T> std::ops::Deref for AdmittedVec<T> {
    type Target = [T];
    fn deref(&self) -> &Self::Target {
        &self.values
    }
}

impl<'a, T> IntoIterator for &'a AdmittedVec<T> {
    type Item = &'a T;
    type IntoIter = std::slice::Iter<'a, T>;
    fn into_iter(self) -> Self::IntoIter {
        self.values.iter()
    }
}

impl<T> AdmittedVec<T> {
    /// A known array layout, admitted before its fallible allocation.
    /// # Errors
    /// Returns layout overflow, admission refusal or allocator failure.
    pub fn with_capacity(
        capacity: usize,
        workspace: &WorkspaceCapability,
    ) -> Result<Self, EvalError> {
        let layout = std::alloc::Layout::array::<T>(capacity)
            .map_err(|_| EvalError::WorkspaceBoundOverflow)?;
        let allocation = if layout.size() == 0 {
            None
        } else {
            Some(workspace.charge(
                u64::try_from(layout.size()).map_err(|_| EvalError::WorkspaceBoundOverflow)?,
            )?)
        };
        let values = vector::<T>(capacity, "admitted vector")?;
        Ok(Self {
            values,
            allocation,
            workspace: workspace.clone(),
        })
    }

    /// Mutating existing elements cannot grow or detach the covered buffer.
    pub fn as_mut_slice(&mut self) -> &mut [T] {
        &mut self.values
    }

    /// Empty native storage sharing its execution's growth capability.
    #[must_use]
    pub fn new(workspace: &WorkspaceCapability) -> Self {
        Self {
            values: Vec::new(),
            allocation: None,
            workspace: workspace.clone(),
        }
    }

    /// Transfer a buffer and its existing admission together after an in-place
    /// representation change. Callers must retain the original allocation grant.
    pub(crate) fn from_parts(
        values: Vec<T>,
        allocation: Option<WorkspaceAllocation>,
        workspace: &WorkspaceCapability,
    ) -> Self {
        Self {
            values,
            allocation,
            workspace: workspace.clone(),
        }
    }

    pub(crate) fn into_parts(self) -> (Vec<T>, Option<WorkspaceAllocation>) {
        (self.values, self.allocation)
    }

    /// Move resident storage without copying its elements. A bounded vector
    /// returns its complete unchanged owner rather than shedding admission.
    ///
    /// # Errors
    /// Returns the original admitted owner for a bounded execution.
    pub fn try_into_resident(self) -> Result<Vec<T>, Self> {
        if self.workspace.is_bounded() {
            return Err(self);
        }
        Ok(self.values)
    }

    /// Retain a subset in place without growing or detaching the original array.
    pub(crate) fn retain(&mut self, keep: impl FnMut(&T) -> bool) {
        self.values.retain(keep);
    }

    /// Drop all values while retaining the existing metadata buffer admission.
    pub fn clear(&mut self) {
        self.values.clear();
    }
    /// Drop values after a prefix, retaining the existing buffer admission.
    pub fn truncate(&mut self, length: usize) {
        self.values.truncate(length);
    }

    /// Move the last element while retaining its vector metadata admission.
    pub fn pop(&mut self) -> Option<T> {
        self.values.pop()
    }

    /// Remove one ordered element while keeping the original array admitted.
    pub(crate) fn remove(&mut self, index: usize) -> T {
        self.values.remove(index)
    }

    /// Drain a suffix while borrowing the original admitted metadata owner.
    pub fn drain_from(&mut self, start: usize) -> std::vec::Drain<'_, T> {
        self.values.drain(start..)
    }

    /// Append after admitting and fallibly allocating replacement capacity.
    ///
    /// # Errors
    /// Returns layout overflow, admission refusal or allocator failure.
    pub fn push(&mut self, value: T) -> Result<(), EvalError> {
        self.reserve_next()?;
        self.values.push(value);
        Ok(())
    }

    pub(crate) fn try_extend(
        &mut self,
        values: impl IntoIterator<Item = T>,
    ) -> Result<(), EvalError> {
        for value in values {
            self.push(value)?;
        }
        Ok(())
    }

    pub(crate) fn from_resident(values: Vec<T>) -> Self {
        Self {
            values,
            allocation: None,
            workspace: WorkspaceCapability::resident(),
        }
    }

    pub(crate) fn reserve_additional(&mut self, additional: usize) -> Result<(), EvalError> {
        let required = self
            .values
            .len()
            .checked_add(additional)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        if required > self.values.capacity() {
            let capacity = self
                .values
                .capacity()
                .checked_mul(2)
                .ok_or(EvalError::WorkspaceBoundOverflow)?
                .max(required);
            self.workspace
                .reserve_vec(&mut self.values, &mut self.allocation, capacity)?;
        }
        Ok(())
    }

    pub(crate) fn reserve_next(&mut self) -> Result<(), EvalError> {
        if self.values.len() == self.values.capacity() {
            let needed = self
                .values
                .len()
                .checked_add(1)
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            let capacity = self
                .values
                .capacity()
                .checked_mul(2)
                .ok_or(EvalError::WorkspaceBoundOverflow)?
                .max(needed);
            let layout = std::alloc::Layout::array::<T>(capacity)
                .map_err(|_| EvalError::WorkspaceBoundOverflow)?;
            let allocation = self.workspace.charge(
                u64::try_from(layout.size()).map_err(|_| EvalError::WorkspaceBoundOverflow)?,
            )?;
            let mut replacement = vector::<T>(capacity, "admitted vector")?;
            replacement.append(&mut self.values);
            self.values = replacement;
            self.allocation = Some(allocation);
        }
        Ok(())
    }

    pub(crate) fn push_reserved(&mut self, value: T) {
        assert!(
            self.values.len() < self.values.capacity(),
            "native destination reserved before ownership transfer"
        );
        self.values.push(value);
    }
}

/// A fixed-key native table retaining its actual hashbrown array admission.
/// Returned references borrow the table; replacement moves each entry once.
#[derive(Debug)]
pub struct AdmittedMap<K, V> {
    entries: hashbrown::HashTable<(K, V)>,
    allocation: Option<WorkspaceAllocation>,
}

impl<K, V> Default for AdmittedMap<K, V> {
    fn default() -> Self {
        Self {
            entries: hashbrown::HashTable::new(),
            allocation: None,
        }
    }
}

impl<K, V> AdmittedMap<K, V> {
    pub(crate) fn into_parts(self) -> (hashbrown::HashTable<(K, V)>, Option<WorkspaceAllocation>) {
        (self.entries, self.allocation)
    }
}

impl<K: Eq + std::hash::Hash, V> AdmittedMap<K, V> {
    /// Destroy entries while retaining the admitted reusable table buffer.
    pub fn clear(&mut self) {
        self.entries.clear();
    }

    /// The live entry count; retained spare capacity remains admitted.
    #[must_use]
    pub fn len(&self) -> usize {
        self.entries.len()
    }

    /// Whether there are no live entries.
    #[must_use]
    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    /// Copy table metadata through the same admitted insert home. The callback
    /// carries each original payload owner or admits its actual independent copy.
    pub(crate) fn clone_entries_admitted(
        &self,workspace:&WorkspaceCapability,
        mut copy:impl FnMut(&K,&V)->Result<(K,V),EvalError>,
    )->Result<Self,EvalError> {
        let mut output=Self::default();
        for (key,value) in self.iter() {
            let (key,value)=copy(key,value)?;
            output.insert_admitted(key,value,workspace)?;
        }
        Ok(output)
    }

    /// Borrow membership without cloning a key or allocating a table.
    #[must_use]
    pub fn contains_key<Q: ?Sized + Eq + std::hash::Hash>(&self,key:&Q)->bool
    where K:std::borrow::Borrow<Q> { self.get(key).is_some() }

    /// Borrow payloads without a copied iterator buffer.
    pub fn values(&self)->impl Iterator<Item=&V> { self.iter().map(|(_,value)|value) }

    #[cfg(test)]
    pub(crate) fn insert(&mut self,key:K,value:V)->Option<V> {
        self.insert_admitted(key,value,&WorkspaceCapability::resident()).expect("resident table allocation failed")
    }

    /// Borrow each existing key and value without an iterator allocation.
    pub fn iter(&self) -> impl Iterator<Item = (&K, &V)> {
        self.entries.iter().map(|entry| (&entry.0, &entry.1))
    }

    /// Borrow existing keys and mutable values without growing the table.
    pub fn iter_mut(&mut self) -> impl Iterator<Item = (&K, &mut V)> {
        self.entries
            .iter_mut()
            .map(|entry| (&entry.0, &mut entry.1))
    }

    /// Borrow a value using the same fixed hash and a borrowed key identity.
    #[must_use]
    pub fn get<Q: ?Sized + Eq + std::hash::Hash>(&self, key: &Q) -> Option<&V>
    where
        K: std::borrow::Borrow<Q>,
    {
        self.get_key_value(key).map(|(_, value)| value)
    }

    /// Borrow the original key and value without copying either owner.
    #[must_use]
    pub fn get_key_value<Q: ?Sized + Eq + std::hash::Hash>(&self, key: &Q) -> Option<(&K, &V)>
    where
        K: std::borrow::Borrow<Q>,
    {
        self.entries
            .find(purrdf_hash::fixed::hash_one(key), |entry| {
                <K as std::borrow::Borrow<Q>>::borrow(&entry.0) == key
            })
            .map(|entry| (&entry.0, &entry.1))
    }

    /// Mutate a value while its original table and payload owners remain alive.
    #[must_use]
    pub fn get_mut<Q: ?Sized + Eq + std::hash::Hash>(&mut self, key: &Q) -> Option<&mut V>
    where
        K: std::borrow::Borrow<Q>,
    {
        self.entries
            .find_mut(purrdf_hash::fixed::hash_one(key), |entry| {
                <K as std::borrow::Borrow<Q>>::borrow(&entry.0) == key
            })
            .map(|entry| &mut entry.1)
    }

    /// Move a value out without allocating or releasing the live table buffer.
    pub fn remove<Q: ?Sized + Eq + std::hash::Hash>(&mut self, key: &Q) -> Option<V>
    where
        K: std::borrow::Borrow<Q>,
    {
        let entry = self
            .entries
            .find_entry(purrdf_hash::fixed::hash_one(key), |entry| {
                <K as std::borrow::Borrow<Q>>::borrow(&entry.0) == key
            })
            .ok()?;
        Some(entry.remove().0.1)
    }

    /// Insert or replace after admitting any concrete replacement table.
    /// # Errors
    /// Returns layout overflow, admission refusal or allocator failure.
    pub fn insert_admitted(
        &mut self,
        key: K,
        value: V,
        workspace: &WorkspaceCapability,
    ) -> Result<Option<V>, EvalError> {
        let hash = purrdf_hash::fixed::hash_one(&key);
        if let Some(entry) = self.entries.find_mut(hash, |entry| entry.0 == key) {
            return Ok(Some(std::mem::replace(&mut entry.1, value)));
        }
        let needed = self
            .entries
            .len()
            .checked_add(1)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        if needed > self.entries.capacity() {
            let capacity = self
                .entries
                .capacity()
                .checked_mul(2)
                .ok_or(EvalError::WorkspaceBoundOverflow)?
                .max(needed);
            let bytes = purrdf_core::hash::hash_table_allocation_bound::<(K, V)>(capacity)
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            let allocation = workspace
                .charge(u64::try_from(bytes).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
            let mut entries = hashbrown::HashTable::new();
            entries
                .try_reserve(capacity, |entry: &(K, V)| {
                    purrdf_hash::fixed::hash_one(&entry.0)
                })
                .map_err(|_| EvalError::AllocationFailed {
                    construct: "admitted native table",
                })?;
            // Draining moves payloads and does not allocate a temporary buffer.
            // The old table stays admitted until assignment destroys its array.
            for entry in self.entries.drain() {
                entries.insert_unique(purrdf_hash::fixed::hash_one(&entry.0), entry, |entry| {
                    purrdf_hash::fixed::hash_one(&entry.0)
                });
            }
            self.entries = entries;
            self.allocation = Some(allocation);
        }
        self.entries.insert_unique(hash, (key, value), |entry| {
            purrdf_hash::fixed::hash_one(&entry.0)
        });
        Ok(None)
    }
}

/// A native binary heap whose capacity and sorted output retain admission.
/// Growth reserves a replacement before moving the original heap buffer.
#[derive(Debug)]
pub struct AdmittedHeap<T: Ord> {
    values: std::collections::BinaryHeap<T>,
    allocation: Option<WorkspaceAllocation>,
    workspace: WorkspaceCapability,
}

impl<T: Ord> AdmittedHeap<T> {
    /// An empty heap associated with this execution.
    #[must_use]
    pub fn new(workspace: &WorkspaceCapability) -> Self {
        Self {
            values: std::collections::BinaryHeap::new(),
            allocation: None,
            workspace: workspace.clone(),
        }
    }

    /// A concrete element array admitted before allocation.
    /// # Errors
    /// Returns layout overflow, admission refusal or allocator failure.
    pub fn with_capacity(
        capacity: usize,
        workspace: &WorkspaceCapability,
    ) -> Result<Self, EvalError> {
        let (values, allocation) =
            AdmittedVec::<T>::with_capacity(capacity, workspace)?.into_parts();
        Ok(Self {
            values: std::collections::BinaryHeap::from(values),
            allocation,
            workspace: workspace.clone(),
        })
    }

    /// Append after fallibly acquiring any required replacement buffer.
    /// # Errors
    /// Returns layout overflow, admission refusal or allocator failure.
    pub fn push(&mut self, value: T) -> Result<(), EvalError> {
        if self.values.len() == self.values.capacity() {
            let needed = self
                .values
                .len()
                .checked_add(1)
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            let capacity = self
                .values
                .capacity()
                .checked_mul(2)
                .ok_or(EvalError::WorkspaceBoundOverflow)?
                .max(needed);
            let replacement = AdmittedVec::<T>::with_capacity(capacity, &self.workspace)?;
            let (mut values, allocation) = replacement.into_parts();
            // Both old and replacement arrays remain admitted during transfer.
            // These operations move elements; none clones an element payload.
            let mut old = std::mem::take(&mut self.values).into_vec();
            values.append(&mut old);
            self.values = std::collections::BinaryHeap::from(values);
            drop(old);
            self.allocation = allocation;
        }
        self.values.push(value);
        Ok(())
    }

    /// Remove the greatest item without growing or detaching storage.
    pub fn pop(&mut self) -> Option<T> {
        self.values.pop()
    }

    /// Borrow the greatest item.
    #[must_use]
    pub fn peek(&self) -> Option<&T> {
        self.values.peek()
    }

    /// The number of retained items.
    #[must_use]
    pub fn len(&self) -> usize {
        self.values.len()
    }

    /// Whether the heap contains no items.
    #[must_use]
    pub fn is_empty(&self) -> bool {
        self.values.is_empty()
    }

    /// Sort in place and transfer the same array together with its grant.
    #[must_use]
    pub fn into_sorted_vec(self) -> AdmittedVec<T> {
        AdmittedVec::from_parts(
            self.values.into_sorted_vec(),
            self.allocation,
            &self.workspace,
        )
    }
}

/// A consuming iterator retaining its original metadata buffer admission.
#[derive(Debug)]
pub struct AdmittedVecIntoIter<T> {
    values: std::vec::IntoIter<T>,
    _allocation: Option<WorkspaceAllocation>,
}
impl<T> Iterator for AdmittedVecIntoIter<T> {
    type Item = T;
    fn next(&mut self) -> Option<T> {
        self.values.next()
    }
    fn size_hint(&self) -> (usize, Option<usize>) {
        self.values.size_hint()
    }
}
impl<T> IntoIterator for AdmittedVec<T> {
    type Item = T;
    type IntoIter = AdmittedVecIntoIter<T>;
    fn into_iter(self) -> Self::IntoIter {
        AdmittedVecIntoIter {
            values: self.values.into_iter(),
            _allocation: self.allocation,
        }
    }
}

/// Shared execution admission lent to native producers and nested evaluation.
/// Each granted owner retains the original typed account without changing its
/// error type or promoting a borrowed provider to a static lifetime.
#[derive(Debug, Clone, Default)]
pub struct WorkspaceCapability {
    owner: Option<Shared<Box<dyn GrowthAdmission>>>,
}

/// Admission for one allocation owner. Declare it after the payload it covers so
/// the payload dies before the capacity is released. It deliberately cannot be
/// cloned independently of that payload.
#[derive(Debug)]
pub struct WorkspaceAllocation {
    _owner: Option<Box<dyn AllocationOwner>>,
}

/// A copied native term that keeps its allocation admitted while borrowed.
#[derive(Debug)]
pub struct WorkspaceTerm {
    value: purrdf_core::TermValue,
    layout: Option<crate::retained::TermLayout>,
    allocation: WorkspaceAllocation,
}
impl std::ops::Deref for WorkspaceTerm {
    type Target = purrdf_core::TermValue;
    fn deref(&self) -> &Self::Target {
        &self.value
    }
}
impl WorkspaceTerm {
    pub(crate) const fn resident(value: purrdf_core::TermValue) -> Self {
        Self::new(value, WorkspaceAllocation { _owner: None })
    }
    /// Enclose a native term built under the supplied pre-acquired admission.
    /// Producers must acquire that admission before constructing the payload.
    #[must_use]
    pub const fn new(value: purrdf_core::TermValue, allocation: WorkspaceAllocation) -> Self {
        Self {
            value,
            layout: None,
            allocation,
        }
    }
    pub(crate) const fn with_layout(
        value: purrdf_core::TermValue,
        allocation: WorkspaceAllocation,
        layout: crate::retained::TermLayout,
    ) -> Self {
        Self {
            value,
            layout: Some(layout),
            allocation,
        }
    }

    pub(crate) fn layout(
        &self,
        workspace: &WorkspaceCapability,
    ) -> Result<crate::retained::TermLayout, EvalError> {
        self.layout
            .map_or_else(|| crate::retained::term_layout(&self.value, workspace), Ok)
    }
    pub(crate) fn into_parts(self) -> (purrdf_core::TermValue, WorkspaceAllocation) {
        (self.value, self.allocation)
    }
    pub(crate) fn into_resident(self) -> Result<purrdf_core::TermValue, Self> {
        if self.allocation.is_bounded() {
            Err(self)
        } else {
            Ok(self.value)
        }
    }

    /// Assemble already admitted components without duplicating their payloads.
    /// All native boxes are acquired before the original component owners drop.
    pub(crate) fn triple(
        s: Self,
        p: Self,
        o: Self,
        workspace: &WorkspaceCapability,
    ) -> Result<Self, EvalError> {
        let mut bytes = 0u64;
        let mut nodes = 1usize;
        for value in [&s, &p, &o] {
            let layout = value.layout(workspace)?;
            bytes = bytes
                .checked_add(layout.bytes)
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            nodes = nodes
                .checked_add(layout.nodes)
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
        }
        let boxes = std::alloc::Layout::array::<purrdf_core::TermValue>(3)
            .map_err(|_| EvalError::WorkspaceBoundOverflow)?
            .size();
        bytes = bytes
            .checked_add(u64::try_from(boxes).map_err(|_| EvalError::WorkspaceBoundOverflow)?)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        struct Components {
            s: Option<purrdf_core::TermValue>,
            p: Option<purrdf_core::TermValue>,
            o: Option<purrdf_core::TermValue>,
            _s_owner: WorkspaceAllocation,
            _p_owner: WorkspaceAllocation,
            admission: WorkspaceAllocation,
        }
        let (s, s_owner) = s.into_parts();
        let (p, p_owner) = p.into_parts();
        let (o, admission) = o.into_parts();
        let mut components = Components {
            s: Some(s),
            p: Some(p),
            o: Some(o),
            _s_owner: s_owner,
            _p_owner: p_owner,
            admission,
        };
        components.admission.resize(bytes)?;
        let s = purrdf_core::small::try_boxed(components.s.take().expect("owned subject"))
            .map_err(|_| EvalError::AllocationFailed {
                construct: "TRIPLE subject box",
            })?;
        let p = purrdf_core::small::try_boxed(components.p.take().expect("owned predicate"))
            .map_err(|_| EvalError::AllocationFailed {
                construct: "TRIPLE predicate box",
            })?;
        let o = purrdf_core::small::try_boxed(components.o.take().expect("owned object")).map_err(
            |_| EvalError::AllocationFailed {
                construct: "TRIPLE object box",
            },
        )?;
        let value = purrdf_core::TermValue::Triple {
            s: s.into(),
            p: p.into(),
            o: o.into(),
        };
        let value = Self::with_layout(
            value,
            components.admission,
            crate::retained::TermLayout { bytes, nodes },
        );
        Ok(value)
    }
}

impl WorkspaceAllocation {
    /// Whether a live operational account owns this allocation grant.
    #[must_use]
    pub const fn is_bounded(&self) -> bool {
        self._owner.is_some()
    }
    /// Admit a replacement capacity before growth, or settle a temporary peak
    /// to the actual surviving payload. A refused resize preserves the previous
    /// charge and the account's first typed operational failure.
    ///
    /// # Errors
    /// Returns checked-layout overflow or the original admission refusal.
    pub fn resize(&mut self, bytes: u64) -> Result<(), EvalError> {
        self._owner
            .as_mut()
            .map_or(Ok(()), |owner| owner.resize(bytes))
    }
}

impl WorkspaceCapability {
    /// Publish native-authored algebra text with its original immutable owner.
    pub(crate) fn authored_text(
        &self,
        value: &(impl std::fmt::Display + ?Sized),
    ) -> Result<purrdf_lex::allocation::SharedText, EvalError> {
        let length = display_len(value)?;
        let control =
            purrdf_lex::allocation::SharedText::allocation_layout::<WorkspaceAllocation>().size();
        let bytes = length
            .checked_add(control)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        let admission =
            self.charge(u64::try_from(bytes).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
        let text = format_exact(value, length, "authored algebra text")?;
        purrdf_lex::allocation::SharedText::try_from_admitted(text, admission).map_err(|_| {
            EvalError::AllocationFailed {
                construct: "authored algebra text owner",
            }
        })
    }

    /// Compute the fixed RDF identity hash under its native traversal admission.
    ///
    /// # Errors
    /// Returns layout overflow, admission refusal or native allocation failure.
    pub fn term_hash(&self, value: &purrdf_core::TermValue) -> Result<u64, EvalError> {
        use std::hash::Hasher;
        let nodes = crate::retained::term_layout(value, self)?.nodes;
        let bytes = purrdf_core::TermValue::hash_workspace_bound(nodes)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        let _working =
            self.charge(u64::try_from(bytes).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
        let mut hasher = purrdf_hash::fixed::FixedHasher::default();
        value
            .try_hash(&mut hasher)
            .map_err(|_| EvalError::AllocationFailed {
                construct: "native term hash",
            })?;
        Ok(hasher.finish())
    }
    /// Whether allocations admitted by this capability retain an operational account.
    #[must_use]
    pub const fn is_bounded(&self) -> bool {
        self.owner.is_some()
    }

    /// Construct a literal from stable borrowed formatting under one payload grant.
    /// This calls the existing term constructor with owned strings, avoiding
    /// a rendered temporary followed by a second lexical copy.
    /// # Errors
    /// Returns typed checked-layout, admission, allocation or formatting failure.
    pub fn literal(
        &self,
        lexical: impl core::fmt::Display,
        datatype: &str,
    ) -> Result<WorkspaceTerm, EvalError> {
        let length = display_len(&lexical)?;
        let bytes = length
            .checked_add(datatype.len())
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        let allocation =
            self.charge(u64::try_from(bytes).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
        let lexical = format_exact(&lexical, length, "native literal lexical text")?;
        let datatype = string(datatype, "native literal datatype")?;
        Ok(WorkspaceTerm::new(
            purrdf_core::TermValue::typed_literal(lexical, datatype),
            allocation,
        ))
    }
    /// An allocation-free capability for resident execution.
    #[must_use]
    pub const fn resident() -> Self {
        Self { owner: None }
    }

    /// Compare RDF identity while admitting the native comparison work-list.
    ///
    /// # Errors
    /// Returns physical overflow, admission refusal or native allocator refusal.
    pub fn terms_equal(
        &self,
        left: &purrdf_core::TermValue,
        right: &purrdf_core::TermValue,
    ) -> Result<bool, EvalError> {
        self.terms_cmp_inner(left, right, true)
            .map(|order| order == std::cmp::Ordering::Equal)
    }

    /// Compare the total RDF order under the same native work-list admission.
    ///
    /// # Errors
    /// Returns physical overflow, admission refusal or native allocator refusal.
    pub fn terms_cmp(
        &self,
        left: &purrdf_core::TermValue,
        right: &purrdf_core::TermValue,
    ) -> Result<std::cmp::Ordering, EvalError> {
        self.terms_cmp_inner(left, right, false)
    }

    fn terms_cmp_inner(
        &self,
        left: &purrdf_core::TermValue,
        right: &purrdf_core::TermValue,
        equality: bool,
    ) -> Result<std::cmp::Ordering, EvalError> {
        let left_nodes = crate::retained::term_layout(left, self)?.nodes;
        let right_nodes = crate::retained::term_layout(right, self)?.nodes;
        let bytes = purrdf_core::TermValue::comparison_workspace_bound(left_nodes.min(right_nodes))
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        let _temporary =
            self.charge(u64::try_from(bytes).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
        let order = if equality {
            left.try_eq(right).map(|equal| {
                if equal {
                    std::cmp::Ordering::Equal
                } else {
                    std::cmp::Ordering::Less
                }
            })
        } else {
            left.try_cmp(right)
        };
        order.map_err(|_| EvalError::AllocationFailed {
            construct: "native term comparison",
        })
    }
    /// Admit and fallibly replace a native producer's vector capacity. Producer
    /// fields retain `allocation` after the vector they cover and never detach it.
    ///
    /// # Errors
    /// Returns layout overflow, admission refusal or allocator failure before growth.
    pub fn reserve_vec<T>(
        &self,
        values: &mut Vec<T>,
        allocation: &mut Option<WorkspaceAllocation>,
        required: usize,
    ) -> Result<(), EvalError> {
        if required <= values.capacity() {
            return Ok(());
        }
        let layout = std::alloc::Layout::array::<T>(required)
            .map_err(|_| EvalError::WorkspaceBoundOverflow)?;
        let next = self
            .charge(u64::try_from(layout.size()).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
        let mut replacement = vector(required, "native producer vector")?;
        replacement.append(values);
        *values = replacement;
        *allocation = Some(next);
        Ok(())
    }

    /// Admit and fallibly replace a native producer's UTF-8 storage capacity.
    ///
    /// # Errors
    /// Returns admission refusal or allocator failure before any buffer growth.
    pub fn reserve_string(
        &self,
        value: &mut String,
        allocation: &mut Option<WorkspaceAllocation>,
        required: usize,
    ) -> Result<(), EvalError> {
        if required <= value.capacity() {
            return Ok(());
        }
        let layout = std::alloc::Layout::array::<u8>(required)
            .map_err(|_| EvalError::WorkspaceBoundOverflow)?;
        let next = self
            .charge(u64::try_from(layout.size()).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
        let mut replacement = String::new();
        replacement
            .try_reserve_exact(required)
            .map_err(|_| EvalError::AllocationFailed {
                construct: "native producer string",
            })?;
        replacement.push_str(value);
        *value = replacement;
        *allocation = Some(next);
        Ok(())
    }
    /// Copy a term through the first-party fallible native clone. Its immutable
    /// carrier preserves RDF identity and retains payload admission after return.
    ///
    /// # Errors
    /// Returns checked physical overflow, admission refusal or allocator failure.
    pub fn clone_term(&self, value: &purrdf_core::TermValue) -> Result<WorkspaceTerm, EvalError> {
        let layout = crate::retained::term_layout(value, self)?;
        let temporary = purrdf_core::TermValue::clone_workspace_bound(layout.nodes)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        let temporary =
            self.charge(u64::try_from(temporary).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
        let allocation = self.charge(layout.bytes)?;
        let value = value.try_clone().map_err(|_| EvalError::AllocationFailed {
            construct: "native term clone",
        })?;
        drop(temporary);
        Ok(WorkspaceTerm::with_layout(value, allocation, layout))
    }
    pub(crate) fn has_failed(&self) -> bool {
        self.owner.as_ref().is_some_and(|owner| owner.has_failed())
    }
    /// Admit output staged across context destruction. The final publication
    /// settles its actual retained layout before it releases execution capacity.
    pub(crate) fn retain(&self, bytes: u64) -> Result<(), EvalError> {
        self.owner
            .as_ref()
            .map_or(Ok(()), |owner| owner.retain(bytes))
    }
    /// Reserve capacity before constructing or growing a native owner.
    ///
    /// # Errors
    /// Returns overflow or the first operational refusal, before allocation.
    pub fn charge(&self, bytes: u64) -> Result<WorkspaceAllocation, EvalError> {
        if self.has_failed() {
            return Err(EvalError::WorkspaceStopped);
        }
        match &self.owner {
            Some(owner) => owner.charge(bytes),
            None => Ok(WorkspaceAllocation { _owner: None }),
        }
    }
}

struct GrowthOwner<E> {

impl<E: Clone + std::error::Error + Send + Sync + 'static> GrowthAdmission for [GrowthOwner<E>; 1] {
    fn has_failed(&self) -> bool {
        self[0].has_failed()
    }
    fn charge(&self, bytes: u64) -> Result<WorkspaceAllocation, EvalError> {
        self[0].charge(bytes)
    }
    fn retain(&self, bytes: u64) -> Result<(), EvalError> {
        self[0].retain(bytes)
    }
}

impl<E: Clone + std::error::Error + Send + Sync + 'static> GrowthAdmission for GrowthOwner<E> {
    fn has_failed(&self) -> bool {
        lock(&self.account).failed
    }
    fn retain(&self, bytes: u64) -> Result<(), EvalError> {
        let mut account = lock(&self.account);
        if account.failed {
            return Err(EvalError::WorkspaceStopped);
        }
        let base = account
            .base
            .checked_add(bytes)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        let total = base
            .checked_add(account.children)
            .and_then(|bytes| bytes.checked_add(account.control))
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        if let Err(error) = account.reservation.resize(total) {
            account.failure = Some(error);
            account.failed = true;
            return Err(EvalError::WorkspaceStopped);
        }
        account.base = base;
        Ok(())
    }
    fn charge(&self, bytes: u64) -> Result<WorkspaceAllocation, EvalError> {
        let owner_bytes = u64::try_from(size_of::<WorkspaceCharge<E>>())
            .map_err(|_| EvalError::WorkspaceBoundOverflow)?;
        let bytes = bytes
            .checked_add(owner_bytes)
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        let workspace = QueryWorkspace {
            account: Some(self.account.clone()),
            capability: WorkspaceCapability::default(),
        };
        let charge = workspace.charge(bytes)?;
        // The concrete box layout was admitted along with the payload above.
        let charge =
            purrdf_core::small::try_boxed_one(charge).map_err(|_| EvalError::AllocationFailed {
                construct: "workspace allocation owner",
            })?;
        Ok(WorkspaceAllocation {
            _owner: Some(charge),
        })
    }
}

struct Account<E> {
    reservation: OwnedWorkspaceReservation<E>,
    base: u64,
    control: u64,
    children: u64,
    failure: Option<E>,
    failed: bool,
}

/// Resident execution has no account; operational execution shares its actual
/// owned reservation, including the first failure a non-latching adapter returns.
pub(crate) struct QueryWorkspace<E> {
    account: Option<Shared<Mutex<Account<E>>>>,
    capability: WorkspaceCapability,
}

impl<E> Clone for QueryWorkspace<E> {
    fn clone(&self) -> Self {
        Self {
            account: self.account.clone(),
            capability: self.capability.clone(),
        }
    }
}

impl<E> std::fmt::Debug for QueryWorkspace<E> {
    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {

fn lock<E>(account: &Mutex<Account<E>>) -> MutexGuard<'_, Account<E>> {
    account
        .lock()
        .unwrap_or_else(std::sync::PoisonError::into_inner)
}

impl<E> QueryWorkspace<E> {
    pub(crate) fn control_bytes() -> Result<u64, EvalError> {
        fallible_shared_layout::<Mutex<Account<E>>>()?
            .checked_add(fallible_shared_layout::<Box<dyn GrowthAdmission>>()?)
            .and_then(|bytes| bytes.checked_add(u64::try_from(size_of::<GrowthOwner<E>>()).ok()?))
            .ok_or(EvalError::WorkspaceBoundOverflow)
    }
    pub(crate) const fn resident() -> Self {
        Self {
            account: None,
            capability: WorkspaceCapability { owner: None },
        }
    }

    pub(crate) fn owned(
        reservation: OwnedWorkspaceReservation<E>,
        bytes: u64,
    ) -> Result<Self, EvalError>
    where
        E: Clone + std::error::Error + Send + Sync + 'static,
    {
        let control = Self::control_bytes()?;
        // The factory caller reserves this complete layout before construction;
        // control is fixed across base resizing and belongs to the final account.
        let account = Shared::try_new(Mutex::new(Account {
            reservation,
            base: bytes,
            control,
            children: 0,
            failure: None,
            failed: false,
        }))
        .map_err(|_| EvalError::AllocationFailed {
            construct: "workspace account",
        })?;
        let owner = purrdf_core::small::try_boxed_one(GrowthOwner {
            account: account.clone(),
        })
        .map_err(|_| EvalError::AllocationFailed {
            construct: "workspace capability owner",
        })?;
        let owner: Box<dyn GrowthAdmission> = owner;
        let owner = Shared::try_new(owner).map_err(|_| EvalError::AllocationFailed {
            construct: "workspace capability",
        })?;
        Ok(Self {
            account: Some(account),
            capability: WorkspaceCapability { owner: Some(owner) },
        })
    }

    pub(crate) const fn is_bounded(&self) -> bool {
        self.account.is_some()
    }
}

/// Actual first-party shared header and payload, including alignment.
pub(crate) fn fallible_shared_layout<T>() -> Result<u64, EvalError> {
    u64::try_from(Shared::<T>::allocation_layout().size())
        .map_err(|_| EvalError::WorkspaceBoundOverflow)
}

/// Actual std Arc layout for resident/shared owners not yet at the fallible home.
pub(crate) fn shared_layout<T>() -> Result<u64, EvalError> {
    let (layout, _) = std::alloc::Layout::new::<[usize; 2]>()
        .extend(std::alloc::Layout::new::<T>())
        .map_err(|_| EvalError::WorkspaceBoundOverflow)?;
    u64::try_from(layout.pad_to_align().size()).map_err(|_| EvalError::WorkspaceBoundOverflow)
}

impl<E: Clone> QueryWorkspace<E> {
    pub(crate) fn source_error(&self, error: E) -> EvalError
    where
        E: std::fmt::Display,
    {
        if let Some(account) = &self.account {
            let mut account = lock(account);
            if !account.failed {
                account.failure = Some(error);
                account.failed = true;
            }
            EvalError::WorkspaceStopped
        } else {
            EvalError::source_read(error)
        }
    }

    pub(crate) fn take_failure(&self) -> Option<E> {
        self.account
            .as_ref()
            .and_then(|account| lock(account).failure.take())
    }

    #[cfg(test)]
    pub(crate) fn has_failed(&self) -> bool {
        self.account
            .as_ref()
            .is_some_and(|account| lock(account).failed)
    }

    /// Admit a new owner before it allocates. Its charge cannot be released before
    /// the allocation itself: callers place the charge after the covered payload.
    pub(crate) fn charge(&self, bytes: u64) -> Result<WorkspaceCharge<E>, EvalError>
    where
        E: std::error::Error + Send + Sync + 'static,
    {
        if let Some(account) = &self.account {
            let mut account = lock(account);
            if account.failed {
                return Err(EvalError::WorkspaceStopped);
            }
            let children = account
                .children
                .checked_add(bytes)
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            let total = account
                .base
                .checked_add(children)
                .and_then(|bytes| bytes.checked_add(account.control))
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            if let Err(error) = account.reservation.resize(total) {
                account.failure = Some(error);
                account.failed = true;
                return Err(EvalError::WorkspaceStopped);
            }
            account.children = children;
        }
        Ok(WorkspaceCharge {
            workspace: self.clone(),
            bytes,
        })
    }
}

impl<E: Clone + std::error::Error + Send + Sync + 'static> QueryWorkspace<E> {
    pub(crate) fn capability(&self) -> WorkspaceCapability {
        self.capability.clone()
    }
    /// Resize the execution's certified base while preserving independently
    /// admitted child owners. Logical overflow is distinct from a backend refusal.
    pub(crate) fn set_base(&mut self, bytes: u64) -> Result<(), EvalError> {
        if let Some(account) = &self.account {
            let mut account = lock(account);
            if account.failed {
                return Err(EvalError::WorkspaceStopped);
            }
            let total = bytes
                .checked_add(account.children)
                .and_then(|bytes| bytes.checked_add(account.control))
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            if let Err(error) = account.reservation.resize(total) {
                account.failure = Some(error);
                account.failed = true;
                return Err(EvalError::WorkspaceStopped);
            }
            account.base = bytes;
        }
        Ok(())
    }
}

/// One engine-owned allocation's admission. Clone the enclosing payload owner,
/// never this individual charge: sharing one allocation shares one reservation.
pub(crate) struct WorkspaceCharge<E: Clone> {
    workspace: QueryWorkspace<E>,
    bytes: u64,
}

impl<E: Clone> std::fmt::Debug for WorkspaceCharge<E> {
    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        formatter
            .debug_struct("WorkspaceCharge")
            .field("bytes", &self.bytes)
            .finish_non_exhaustive()
    }
}

impl<E: Clone + std::error::Error + Send + Sync + 'static> AllocationOwner for WorkspaceCharge<E> {
    fn resize(&mut self, bytes: u64) -> Result<(), EvalError> {
        let bytes = bytes
            .checked_add(
                u64::try_from(size_of::<Self>()).map_err(|_| EvalError::WorkspaceBoundOverflow)?,
            )
            .ok_or(EvalError::WorkspaceBoundOverflow)?;
        if let Some(account) = &self.workspace.account {
            let mut account = lock(account);
            if account.failed {
                return Err(EvalError::WorkspaceStopped);
            }
            let children = account
                .children
                .checked_sub(self.bytes)
                .expect("an owner resizes only its own charge")
                .checked_add(bytes)
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            let total = account
                .base
                .checked_add(children)
                .and_then(|bytes| bytes.checked_add(account.control))
                .ok_or(EvalError::WorkspaceBoundOverflow)?;
            if let Err(error) = account.reservation.resize(total) {
                account.failure = Some(error);
                account.failed = true;
                return Err(EvalError::WorkspaceStopped);
            }
            account.children = children;
        }
        self.bytes = bytes;
        Ok(())
    }
}

impl<E: Clone> Drop for WorkspaceCharge<E> {
    fn drop(&mut self) {
        if let Some(account) = &self.workspace.account {
            let mut account = lock(account);
            let children = account
                .children
                .checked_sub(self.bytes)
                .expect("workspace owner releases only its own charge");
            let bytes = account
                .base
                .checked_add(children)
                .and_then(|bytes| bytes.checked_add(account.control))
                .expect("shrinking child capacity cannot overflow");
            match account.reservation.resize(bytes) {
                Ok(()) => account.children = children,
                Err(error) => {
                    if !account.failed {
                        account.failure = Some(error);
                        account.failed = true;
                    }
                }
            }
        }
    }
}

/// Fallible vector capacity creation, after the owner has admitted its layout.
pub(crate) fn vector<T>(capacity: usize, construct: &'static str) -> Result<Vec<T>, EvalError> {
    let mut vector = Vec::new();
    vector
        .try_reserve_exact(capacity)
        .map_err(|_| EvalError::AllocationFailed { construct })?;
    Ok(vector)
}

pub(crate) fn string(value: &str, construct: &'static str) -> Result<String, EvalError> {
    let mut string = String::new();
    string
        .try_reserve_exact(value.len())
        .map_err(|_| EvalError::AllocationFailed { construct })?;
    string.push_str(value);
    Ok(string)
}

/// Measure a Display value without an output allocation, then admit and render
/// into one fallibly reserved buffer. Display implementations used here render
/// borrowed diagnostic fields; the two passes must have identical output.
pub(crate) fn display_len(value: &(impl core::fmt::Display + ?Sized)) -> Result<usize, EvalError> {
    use core::fmt::Write;
    struct Counter(usize);
    impl Write for Counter {
        fn write_str(&mut self, text: &str) -> core::fmt::Result {
            self.0 = self.0.checked_add(text.len()).ok_or(core::fmt::Error)?;
            Ok(())
        }
    }
    let mut counter = Counter(0);
    write!(&mut counter, "{value}").map_err(|_| EvalError::WorkspaceBoundOverflow)?;
    Ok(counter.0)
}

pub(crate) fn format_exact(
    value: &(impl core::fmt::Display + ?Sized),
    length: usize,
    construct: &'static str,
) -> Result<String, EvalError> {
    use core::fmt::Write;
    let mut text = String::new();
    text.try_reserve_exact(length)
        .map_err(|_| EvalError::AllocationFailed { construct })?;
    struct Bounded<'a> {
        text: &'a mut String,
        length: usize,
    }
    impl Write for Bounded<'_> {
        fn write_str(&mut self, text: &str) -> core::fmt::Result {
            if self
                .text
                .len()
                .checked_add(text.len())
                .is_none_or(|n| n > self.length)
            {
                return Err(core::fmt::Error);
            }
            self.text.push_str(text);
            Ok(())
        }
    }
    write!(
        &mut Bounded {
            text: &mut text,
            length
        },
        "{value}"
    )
    .map_err(|_| EvalError::UnstableNativeDiagnostic)?;
    if text.len() != length {
        return Err(EvalError::UnstableNativeDiagnostic);
    }
    Ok(text)
}

pub(crate) fn render(
    workspace: &WorkspaceCapability,
    value: &EvalError,
    construct: &'static str,
) -> Result<(String, WorkspaceAllocation), EvalError> {
    let length = display_len(value)?;
    let admission =
        workspace.charge(u64::try_from(length).map_err(|_| EvalError::WorkspaceBoundOverflow)?)?;
    let text = format_exact(value, length, construct)?;
    Ok((text, admission))
}

    /// Deliberately non-latching backing failure: the engine must retain the
    /// original typed cause without cloning or formatting it.
    struct GroundFaultView<'a> {
        inner: &'a purrdf_core::RdfDataset,
        failed_id: TermId,
        clones: Arc<AtomicUsize>,
    }

    impl purrdf_core::DatasetView for GroundFaultView<'_> {
        type Id = TermId;
        type ReadError = Refusal;
        type TermGuard<'a>
            = purrdf_core::TermRef<'a>
        where
            Self: 'a;
        type ProbePlan = purrdf_core::QuadProbePlan;

        fn quads(&self) -> impl Iterator<Item = purrdf_core::QuadIds> + '_ {
            purrdf_core::DatasetView::quads(self.inner)
        }
        fn resolve(&self, id: TermId) -> Result<Self::TermGuard<'_>, Refusal> {
            if id == self.failed_id {
                Err(Refusal {
                    identity: 41,
                    clones: self.clones.clone(),
                })
            } else {
                Ok(self.inner.resolve(id))
            }
        }
        fn term_id_by_value(
            &self,
            value: &purrdf_core::TermValue,
        ) -> Result<Option<TermId>, Refusal> {
            Ok(self.inner.term_id_by_value(value))
        }
        fn capabilities(&self) -> purrdf_core::RdfStoreCapabilities {
            purrdf_core::DatasetView::capabilities(self.inner)
        }
        fn probe_plan(
            &self,
            s: bool,
            p: bool,
            o: bool,
            g: purrdf_core::GraphMatch,
        ) -> Self::ProbePlan {
            purrdf_core::DatasetView::probe_plan(self.inner, s, p, o, g)
        }
        fn quads_for_pattern_with_plan(
            &self,
            plan: &Self::ProbePlan,
            s: Option<TermId>,
            p: Option<TermId>,
            o: Option<TermId>,
            g: purrdf_core::GraphMatch,
        ) -> impl Iterator<Item = purrdf_core::QuadIds> + '_ {
            purrdf_core::DatasetView::quads_for_pattern_with_plan(self.inner, plan, s, p, o, g)
        }
        fn term_count(&self) -> u64 {
            purrdf_core::DatasetView::term_count(self.inner)
        }
    }

    #[test]
    fn native_prebinding_all_four_forms_keep_original_ast_admission_until_drop() {
        use purrdf_sparql_algebra::SparqlParser;
        let queries = [
            "SELECT ?this WHERE { ?this <http://example.org/p> ?o OPTIONAL { ?this <http://example.org/q> ?v } }",
            "ASK { ?this <http://example.org/p> ?o FILTER(EXISTS { ?this <http://example.org/q> ?v MINUS { ?v <http://example.org/r> ?this } }) }",
            "CONSTRUCT { ?this <http://example.org/p> ?o } WHERE { { SELECT ?o WHERE { ?this <http://example.org/p> ?o } } }",
            "DESCRIBE ?this WHERE { ?this <http://example.org/p> ?o }",
        ];
        for text in queries {
            // The authored fixture and its input value are caller-owned, outside the
            // fresh query workspace. The native AST copy itself uses original Memory.
            let parsed = SparqlParser::new()
                .with_prebound_variables(["this"])
                .parse_query(text)
                .expect("original query form");
            let values = [(
                "this".to_owned(),
                purrdf_core::TermValue::Blank {
                    label: "focus".to_owned(),
                    scope: None,
                },
            )];
            let (workspace, ledger, base) = account(u64::MAX);
            let capability = workspace.capability();
            let mut parse_frame = super::LexicalFrame::new(&capability);
            let (query, original) = {
                let mut memory = purrdf_lex::allocation::Memory::new(&mut parse_frame);
                let query = parsed
                    .clone_with_memory(&mut memory)
                    .expect("native owned AST copy");
                (query, memory.live_bytes())
            };
            assert!(original > 0);
            let live_before_adoption = ledger.live.load(Ordering::Relaxed);
            let original_allocation = parse_frame.into_allocation().expect("original query grant");
            let mut frame =
                super::LexicalFrame::from_allocation(&capability, original_allocation, original);
            assert_eq!(
                ledger.live.load(Ordering::Relaxed),
                live_before_adoption,
                "adoption neither recharges nor resets the live AST"
            );
            let rewritten = {
                let mut memory = purrdf_lex::allocation::Memory::resume(&mut frame, original);
                let query = crate::substitute::apply_shacl_prebinding_with_memory(
                    query,
                    crate::substitute::Prebindings::Owned(&values),
                    &mut memory,
                )
                .expect("complete native prebinding");
                assert!(memory.live_bytes() >= original);
                query
            };
            assert!(ledger.live.load(Ordering::Relaxed) > base);
            drop(rewritten);
            // The explicit ordered query publication still owns the original frame
            // until payload destruction completes; dropping the frame then frees it.
            drop(frame);
            drop(capability);
            drop(workspace);
            assert_eq!(ledger.live.load(Ordering::Relaxed), 0, "form {text}");
            assert_eq!(ledger.clones.load(Ordering::Relaxed), 0);
        }
    }

    fn native_soundness_refusal_is_typed_and_shallow_edges_need_no_heap() {
        use crate::governor::soundness::{
            NodeAnalysisTable, analyze_pattern_admitted, child_edges_admitted,
            plan_cap_pushdown_admitted,
        };
        use purrdf_sparql_algebra::{Query, SparqlParser};
        let query = SparqlParser::new().parse_query(
            "ASK { ?s <http://example.org/p> ?o FILTER(BOUND(?s)) }",
        ).expect("caller-owned shallow fixture");
        let Query::Ask { pattern, .. } = &query else { unreachable!("ASK fixture"); };
        let base = QueryWorkspace::<Refusal>::control_bytes().expect("actual account control");
        let (workspace, ledger, _) = account(base);
        let capability = workspace.capability();
        let edges = child_edges_admitted(pattern, &capability)
            .expect("one classified child stays in the original inline edge slots");
        assert_eq!(ledger.live.load(Ordering::Relaxed), base);
        let mut table = NodeAnalysisTable::default();
        let failure = analyze_pattern_admitted(pattern, &mut table, &capability)
            .expect_err("first native analysis backing storage is refused before growth");
        assert!(matches!(failure, EvalError::WorkspaceStopped));
        assert!(table.is_empty());
        assert_eq!(workspace.take_failure().expect("original source refusal").identity, 7);
        assert_eq!(ledger.live.load(Ordering::Relaxed), base);
        drop(edges);
        drop(capability);
        drop(workspace);
        assert_eq!(ledger.live.load(Ordering::Relaxed), 0);

        let (workspace, ledger, _) = account(base);
        let capability = workspace.capability();
        let failure = plan_cap_pushdown_admitted(pattern, Some(3), &capability)
            .expect_err("the original cap worklist is refused before its first allocation");
        assert!(matches!(failure, EvalError::WorkspaceStopped));
        assert_eq!(workspace.take_failure().expect("same first refusal").identity, 7);
        assert_eq!(ledger.live.load(Ordering::Relaxed), base);
        drop(capability);
        drop(workspace);
        assert_eq!(ledger.live.load(Ordering::Relaxed), 0);
    }

    #[test]
    fn native_correlated_rows_and_maps_keep_original_grants_until_last_owner() {
        use purrdf_sparql_algebra::{Child, Expression, GraphPattern, GroundTerm, NamedNode, Variable};
        let pattern=GraphPattern::Filter {
            expr:Expression::Exists(Child::new(GraphPattern::Bgp { patterns:Vec::new() })),
            inner:Child::new(GraphPattern::Bgp { patterns:Vec::new() }),
        };
        let (workspace,ledger,base)=account(1_000_000);
        let capability=workspace.capability();
        let copied=crate::expr::CorrelatedPattern::copy(&pattern,&capability).expect("native original copy");
        let row=crate::expr::SubstitutionRow::build(&capability,|expr,term,memory| {
            let variable=Variable::from_admitted(capability.authored_text("outer")?);
            let iri=NamedNode::from_admitted(capability.authored_text("http://example.org/native-original")?);
            memory.push(expr,(variable.clone(),Expression::NamedNode(iri.clone())))
                .map_err(|error|crate::expr::correlated_storage_error(memory,error))?;
            memory.push(term,(variable,GroundTerm::NamedNode(iri)))
                .map_err(|error|crate::expr::correlated_storage_error(memory,error))
        }).expect("native immutable row");
        let after_row=ledger.live.load(Ordering::Relaxed);
        let retained=row.clone();
        assert_eq!(ledger.live.load(Ordering::Relaxed),after_row,
            "ordinary row clone retains the original payload and child grant");
        let wrapped=copied.first().expect("move beneath original first-witness wrapper");
        let mut map=crate::expr::SubstitutionSourceMap::default();
        crate::enf::map_clone_admitted(&pattern,
            match wrapped.as_ref() { GraphPattern::Slice { inner,.. }=>inner,_=>panic!("wrapper") },
            &mut map,&capability).expect("actual native source mapping");
        let original_address=std::ptr::from_ref(&pattern) as usize;
        assert!(map.iter().any(|(_,entry)|entry.source==original_address));
        let map=super::SharedWorkspace::new_admitted(map,&capability).expect("original map control");
        let mut stack=crate::expr::CorrelatedMaps::default();
        stack.push(map,&capability).expect("original window");
        let fork=stack.clone();
        drop(stack);
        assert_eq!(fork.iter().count(),1);
        assert!(!retained.term.is_empty());
        drop(wrapped);
        drop(row);
        assert!(ledger.live.load(Ordering::Relaxed)>base);
        drop(capability);
        drop(workspace);
        assert!(ledger.live.load(Ordering::Relaxed)>0);
        drop(retained);
        drop(fork);
        assert_eq!(ledger.live.load(Ordering::Relaxed),0);
    }

    #[test]
    fn native_correlated_preparation_refusal_moves_first_cause_and_releases_partial_tree() {
        use purrdf_sparql_algebra::{Child, Expression, GraphPattern, Literal, Variable};
        let mut pattern=GraphPattern::Bgp { patterns:Vec::new() };
        for index in 0..64 {
            pattern=GraphPattern::Extend {
                inner:Child::new(pattern),
                variable:Variable::new(format!("column{index}")),
                expression:Expression::Literal(Literal::new_simple("original lexical leaf")),
            };
        }
        let base=QueryWorkspace::<Refusal>::control_bytes().expect("account layout");
        for extra in [0,1,64,256,1024] {
            let (workspace,ledger,_)=account(base+extra);
            let capability=workspace.capability();
            let error=crate::eval::PreparedExists::build_admitted(&pattern,&capability)
                .err().expect("the actual native AST cannot fit this original account");
            assert!(matches!(error,EvalError::WorkspaceStopped));
            drop(error);
            assert_eq!(ledger.live.load(Ordering::Relaxed),base,
                "partial native trees, source tables and analysis owners die on failure");
            let cause=workspace.take_failure().expect("original typed source refusal");
            assert_eq!(cause.identity,7);
            assert_eq!(ledger.clones.load(Ordering::Relaxed),0);
            assert!(workspace.take_failure().is_none(),"no copied or replaced provider cause");
            drop(capability);
            drop(workspace);
            assert_eq!(ledger.live.load(Ordering::Relaxed),0);
        }
    }

    #[test]
    fn native_scope_preflight_failure_leaves_flags_maps_and_placeholders_unmodified() {
        let dataset=purrdf_core::RdfDatasetBuilder::new().freeze().expect("empty source");
        let base=QueryWorkspace::<Refusal>::control_bytes().expect("account layout");
        let (workspace,ledger,_)=account(base+1);
        let capability=workspace.capability();

        assert_eq!(ledger.live.load(Ordering::Relaxed),base);
        assert_eq!(workspace.take_failure().expect("original cause").identity,7);
        drop(ctx);
        drop(capability);
        drop(workspace);
        assert_eq!(ledger.live.load(Ordering::Relaxed),0);
    }
