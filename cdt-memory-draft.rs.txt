//! Physical allocation callbacks for the one composite scanner and order walk.
//! The closed leaf borrows a caller's storage owner; it never imports its engine.

use alloc::boxed::Box;
use alloc::string::String;
use alloc::vec::Vec;
use core::alloc::Layout;
use core::fmt;

use crate::{CdtError, CdtTripleTerm, CdtValue};

/// A physical failure, separate from the CDT lexical language.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum StorageError {
    /// A concrete allocation layout or live-byte sum cannot be represented.
    SizeOverflow,
    /// A fallible allocation refused its already admitted destination.
    AllocationFailed,
    /// The caller's admission callback refused; it retains its original typed error.
    AdmissionFailed,
}

impl fmt::Display for StorageError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.write_str(match self {
            Self::SizeOverflow => "composite allocation layout overflow",
            Self::AllocationFailed => "composite destination allocation failed",
            Self::AdmissionFailed => "composite workspace admission refused",
        })
    }
}

/// The leaf's lexical diagnostic or a physical failure.
#[derive(Debug)]
pub enum ReadError {
    /// The existing exact diagnostic, including its original offset and text.
    Lexical(CdtError),
    /// Physical failure; must never be collapsed into a malformed literal.
    Storage(StorageError),
}

impl From<CdtError> for ReadError {
    fn from(error: CdtError) -> Self { Self::Lexical(error) }
}

impl From<StorageError> for ReadError {
    fn from(error: StorageError) -> Self { Self::Storage(error) }
}

impl fmt::Display for ReadError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Lexical(error) => fmt::Display::fmt(error, f),
            Self::Storage(error) => fmt::Display::fmt(error, f),
        }
    }
}

/// A caller's concrete storage owner and fallible native box factory.
///
/// `resize` receives the exact total requested layout of live allocations in this
/// native call, including old and replacement buffers during growth. On parse
/// success, that amount belongs to the returned value and the caller must retain
/// its original admission with it. On failure, the call's locals and diagnostic
/// must die before the caller releases admission. The factories are invoked only
/// after admitting `Layout::new` for their payload; they must allocate exactly that
/// layout fallibly, without an unpriced alternate allocation.
pub trait Storage {
    /// Admit the next live physical byte total before allocation.
    fn resize(&mut self, live_bytes: usize) -> Result<(), StorageError>;
    /// Fallibly allocate the admitted nested composite payload.
    fn boxed_value(&mut self, value: CdtValue) -> Result<Box<CdtValue>, StorageError>;
    /// Fallibly allocate the admitted nested triple payload.
    fn boxed_triple(&mut self, value: CdtTripleTerm) -> Result<Box<CdtTripleTerm>, StorageError>;
}

/// Existing resident APIs' caller-owned allocation adapter.
pub(crate) struct Resident;

impl Storage for Resident {
    fn resize(&mut self, _live_bytes: usize) -> Result<(), StorageError> { Ok(()) }
    fn boxed_value(&mut self, value: CdtValue) -> Result<Box<CdtValue>, StorageError> {
        Ok(Box::new(value))
    }
    fn boxed_triple(&mut self, value: CdtTripleTerm) -> Result<Box<CdtTripleTerm>, StorageError> {
        Ok(Box::new(value))
    }
}

/// One accounting body shared by scanner, measuring renderer and comparison.
pub(crate) struct Memory<'a> {
    storage: &'a mut dyn Storage,
    live: usize,
}

impl<'a> Memory<'a> {
    pub(crate) fn new(storage: &'a mut dyn Storage) -> Self {
        Self { storage, live: 0 }
    }

    pub(crate) const fn live_bytes(&self) -> usize { self.live }

    fn add(&mut self, bytes: usize) -> Result<(), StorageError> {
        let next = self.live.checked_add(bytes).ok_or(StorageError::SizeOverflow)?;
        if next != self.live { self.storage.resize(next)?; }
        self.live = next;
        Ok(())
    }

    fn remove(&mut self, bytes: usize) -> Result<(), StorageError> {
        let next = self.live.checked_sub(bytes).ok_or(StorageError::SizeOverflow)?;
        if next != self.live { self.storage.resize(next)?; }
        self.live = next;
        Ok(())
    }

    pub(crate) fn reserve<T>(
        &mut self, values: &mut Vec<T>, required: usize,
    ) -> Result<(), StorageError> {
        if required <= values.capacity() { return Ok(()); }
        let old = Layout::array::<T>(values.capacity()).map_err(|_| StorageError::SizeOverflow)?.size();
        let next = Layout::array::<T>(required).map_err(|_| StorageError::SizeOverflow)?.size();
        self.add(next)?;
        let mut replacement = Vec::new();
        replacement.try_reserve_exact(required).map_err(|_| StorageError::AllocationFailed)?;
        // Global Vec's exact reservation requests precisely the certified Layout.
        // Refuse a violated allocator/collection certificate before using it.
        if core::mem::size_of::<T>() != 0 && replacement.capacity() != required {
            return Err(StorageError::SizeOverflow);
        }
        replacement.append(values);
        drop(core::mem::replace(values, replacement));
        self.remove(old)
    }

    pub(crate) fn push<T>(&mut self, values: &mut Vec<T>, value: T) -> Result<(), StorageError> {
        let required = values.len().checked_add(1).ok_or(StorageError::SizeOverflow)?;
        if required > values.capacity() {
            let capacity = values.capacity().checked_mul(2).ok_or(StorageError::SizeOverflow)?
                .max(required);
            self.reserve(values, capacity)?;
        }
        values.push(value);
        Ok(())
    }

    pub(crate) fn release_vec<T>(&mut self, values: Vec<T>) -> Result<(), StorageError> {
        let bytes = Layout::array::<T>(values.capacity()).map_err(|_| StorageError::SizeOverflow)?.size();
        drop(values);
        self.remove(bytes)
    }

    pub(crate) fn reserve_string(&mut self, text: &mut String, required: usize) -> Result<(), StorageError> {
        if required <= text.capacity() { return Ok(()); }
        let next = Layout::array::<u8>(required).map_err(|_| StorageError::SizeOverflow)?.size();
        let old = text.capacity();
        self.add(next)?;
        let mut replacement = String::new();
        replacement.try_reserve_exact(required).map_err(|_| StorageError::AllocationFailed)?;
        if replacement.capacity() != required { return Err(StorageError::SizeOverflow); }
        replacement.push_str(text);
        drop(core::mem::replace(text, replacement));
        self.remove(old)
    }

    pub(crate) fn push_char(&mut self, text: &mut String, ch: char) -> Result<(), StorageError> {
        let required = text.len().checked_add(ch.len_utf8()).ok_or(StorageError::SizeOverflow)?;
        if required > text.capacity() {
            let capacity = text.capacity().checked_mul(2).ok_or(StorageError::SizeOverflow)?.max(required);
            self.reserve_string(text, capacity)?;
        }
        text.push(ch);
        Ok(())
    }

    pub(crate) fn string(&mut self, input: &str) -> Result<String, StorageError> {
        let mut result = String::new();
        self.reserve_string(&mut result, input.len())?;
        result.push_str(input);
        Ok(result)
    }

    pub(crate) fn boxed_value(&mut self, value: CdtValue) -> Result<Box<CdtValue>, StorageError> {
        self.add(Layout::new::<CdtValue>().size())?;
        self.storage.boxed_value(value)
    }

    pub(crate) fn boxed_triple(&mut self, value: CdtTripleTerm) -> Result<Box<CdtTripleTerm>, StorageError> {
        self.add(Layout::new::<CdtTripleTerm>().size())?;
        self.storage.boxed_triple(value)
    }
}
