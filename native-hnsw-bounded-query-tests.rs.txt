//! Approximate retrieval driven the way a host drives it: SPARQL query **text**
//! through [`NativeSparqlEngine`], over a real PURREMB artifact.
//!
//! Every other test in this crate opens a cursor directly, or drives the relation through
//! `PropertyFunction::open`. Those pin what the engine emits once it has been asked; what
//! they cannot show is that the seam a host actually reaches through — parse, registry
//! resolution, evaluation, answers — lines up with it. A capability whose only callers
//! construct its internals is a capability no query can reach.
//!
//! The artifact here is the genuine one: a certified source, a typed embedding family, the
//! HNSW guard and payload in the real `INDEX_GUARDS` and `INDEX_PAYLOAD` sections, reopened
//! and verified before the space is built. The engine is a default `NativeSparqlEngine`
//! with no parser options, so the only reason the predicate resolves is the registration.
//!
//! # PurRDF mints no vocabulary
//!
//! `https://example.org/pf#nearest` is a **fixture** IRI supplied by this test in its role
//! as the host. It is not a default, and this crate contains no namespace of its own.

#[path = "support/purremb.rs"]
mod purremb;

#[path = "support/fixture.rs"]
mod fixture;
use fixture::params;
use purrdf_core::term_fixture::empty_dataset as dataset;
use std::sync::Arc;

use purrdf_core::{SparqlRequest, SparqlResult, TermValue};
use purrdf_hnsw::relation::HnswSpace;
use purrdf_sparql_eval::{
    ExtensionEnv, KnnGuard, NativeSparqlEngine, PropertyFunctionRegistry, QueryOptions,
};

/// The caller-supplied predicate this host calls approximate retrieval by.
const NEAREST: &str = "https://example.org/pf#nearest";

/// A space over the real artifact, plus the row terms the fixture bound.
fn space(rows: usize) -> (Arc<HnswSpace>, Vec<TermValue>) {
    let fixture = purremb::Fixture::new(rows, 8, params());
    let guard = KnnGuard::new(rows as u64, rows as u64).expect("valid guard");
    let space = HnswSpace::from_artifact(
        &fixture.bytes,
        fixture.target_set,
        fixture.vector_space,
        fixture.bindings(),
        guard,
    )
    .expect("the artifact yields a space");
    (Arc::new(space), fixture.terms)
}

/// A registry with the relation registered under `NEAREST` — the whole of what a host does.
fn registry(space: &Arc<HnswSpace>) -> PropertyFunctionRegistry {
    let mut registry = PropertyFunctionRegistry::new();
    registry.register(NEAREST.to_owned(), Arc::new(space.relation()));
    registry
}

/// Evaluate `query` with `registry` in scope and return the solution rows as terms.
fn answer(registry: &PropertyFunctionRegistry, query: &str) -> Vec<Vec<Option<TermValue>>> {
    let data = dataset();
    let result = NativeSparqlEngine::new()
        .query_with_options_view(
            &data,
            SparqlRequest {
                query,
                base_iri: None,
                substitutions: &[],
            },
            QueryOptions::new().with_env(
                &ExtensionEnv::over_relations(registry.clone())
                    .expect("the fixture declarations read cleanly"),
            ),
        )
        .unwrap_or_else(|error| panic!("the query must evaluate: {error}"));
    let SparqlResult::Solutions { rows, .. } = result else {
        panic!("a SELECT answers with solutions, got {result:?}");
    };
    rows.iter().map(Clone::clone).collect()
}

#[test]
fn a_sparql_query_retrieves_neighbours_through_the_registered_predicate() {
    let (space, terms) = space(48);
    let seed = &terms[0];
    let rows = answer(
        &registry(&space),
        &format!(
            "SELECT ?neighbour ?distance WHERE {{ \
             ?neighbour <{NEAREST}> ( <{}> 5 ?distance ) }}",
            match seed {
                TermValue::Iri(iri) => iri.clone(),
                other => panic!("the fixture binds IRIs, got {other:?}"),
            }
        ),
    );

    assert_eq!(rows.len(), 5, "the query asked for five neighbours");
    assert!(
        rows.iter().all(|row| row.len() == 2),
        "each solution binds ?neighbour and ?distance"
    );
    assert_eq!(
        rows[0][0].as_ref(),
        Some(seed),
        "a seed is its own nearest neighbour at distance zero"
    );
    assert!(
        rows.iter().all(|row| row[0].is_some() && row[1].is_some()),
        "no solution leaves a projected variable unbound"
    );
}

#[test]
fn the_offer_is_ordered_nearest_first() {
    let (space, terms) = space(64);
    let TermValue::Iri(seed) = &terms[7] else {
        panic!("the fixture binds IRIs");
    };
    let rows = answer(
        &registry(&space),
        &format!(
            "SELECT ?neighbour ?distance WHERE {{ \
             ?neighbour <{NEAREST}> ( <{seed}> 6 ?distance ) }}"
        ),
    );

    let distances: Vec<f64> = rows
        .iter()
        .map(|row| match row[1].as_ref() {
            Some(TermValue::Literal { lexical_form, .. }) => {
                lexical_form.parse().expect("a numeric distance")
            }
            other => panic!("?distance must bind a literal, got {other:?}"),
        })
        .collect();

    assert_eq!(distances.len(), 6);
    assert!(
        distances.windows(2).all(|pair| pair[0] <= pair[1]),
        "the offer must be ordered nearest first, got {distances:?}"
    );
}

#[test]
fn a_limit_takes_a_prefix_of_the_same_offer() {
    // The engine may push a row ceiling down through `open`. That ceiling decides how many
    // rows are emitted and must never decide which graph is searched.
    let (space, terms) = space(64);
    let TermValue::Iri(seed) = &terms[3] else {
        panic!("the fixture binds IRIs");
    };
    let registry = registry(&space);
    let full = answer(
        &registry,
        &format!("SELECT ?neighbour WHERE {{ ?neighbour <{NEAREST}> ( <{seed}> 6 ?distance ) }}"),
    );
    let limited = answer(
        &registry,
        &format!(
            "SELECT ?neighbour WHERE {{ ?neighbour <{NEAREST}> ( <{seed}> 6 ?distance ) }} \
             LIMIT 2"
        ),
    );

    assert_eq!(limited.len(), 2);
    assert_eq!(
        limited,
        full[..2],
        "a LIMIT must yield a prefix of the unlimited offer"
    );
}

#[test]
fn the_rows_come_from_the_registration_and_from_nothing_else() {
    // The non-vacuity control. Without it, every assertion above would be equally satisfied
    // by an engine that matched the pattern some other way, and the tests could not
    // distinguish "retrieval ran" from "the pattern happened to match".
    //
    // The dataset is empty and the predicate is a fixture IRI no quad mentions, so with an
    // empty registry the same query text is an ordinary triple pattern and answers nothing.
    // The only difference between the two runs is the registration.
    let (space, terms) = space(48);
    let TermValue::Iri(seed) = &terms[0] else {
        panic!("the fixture binds IRIs");
    };
    let query = format!("SELECT ?neighbour WHERE {{ ?neighbour <{NEAREST}> ( <{seed}> 5 ?d ) }}");

    let unregistered = answer(&PropertyFunctionRegistry::new(), &query);
    assert!(
        unregistered.is_empty(),
        "an unregistered predicate is an ordinary pattern over an empty dataset, so it \
         cannot be the source of any row"
    );

    let registered = answer(&registry(&space), &query);
    assert_eq!(
        registered.len(),
        5,
        "registering the relation is what produces the rows"
    );
}


/// Real host routes over a persisted empty dataset; the caller-owned index is
/// deliberately outside the measurement, while all per-query native scratch is inside.
mod bounded_native_workspace {
    use super::*;
    use purrdf_alloc_probe::CurrentThreadWindow;
    use purrdf_core::{
        SegmentedBuildLimits, SegmentedBuilder, SegmentedError, SegmentedImage,
        SegmentedReadLimits, SegmentedSession,
    };
    use purrdf_hnsw::relation::HnswObservations;
    use purrdf_sparql_eval::{
        FallibleSparqlError, PreparedQuery, QueryGovernors, RetainedSparqlResult,
    };

    #[global_allocator]
    static GLOBAL: purrdf_alloc_probe::CountingAllocator = purrdf_alloc_probe::CountingAllocator;

    const CEILING: u64 = 8_000_000;

    #[derive(Clone, Copy)]
    enum Route {
        Text,
        Prepared,
        GovernedPrepared,
    }

    const ROUTES: [Route; 3] = [Route::Text, Route::Prepared, Route::GovernedPrepared];

    fn image() -> SegmentedImage {
        SegmentedBuilder::new(SegmentedBuildLimits::new(8, 512, 8, 512, 4).unwrap())
            .seal()
            .unwrap()
    }

    fn open(image: &SegmentedImage, ceiling: u64) -> SegmentedSession {
        // A small real request ledger prevents source-opening capacity from hiding
        // an unpriced native traversal behind a much larger unused request array.
        SegmentedSession::open(
            Arc::new(image.provider()),
            image.receipt(),
            SegmentedReadLimits::new(ceiling, 2, 32, 1_000_000, 8),
        )
        .unwrap()
    }

    fn run(
        engine: &NativeSparqlEngine,
        source: &SegmentedSession,
        query: &str,
        prepared: &PreparedQuery,
        options: QueryOptions<'_>,
        route: Route,
    ) -> RetainedSparqlResult {
        match route {
            Route::Text => engine
                .query_fallible_view(
                    source,
                    SparqlRequest { query, base_iri: None, substitutions: &[] },
                    options,
                )
                .unwrap_or_else(|error| panic!("healthy bounded HNSW text query: {error:?}"))
                .result,
            Route::Prepared => engine
                .query_prepared_fallible_view(source, prepared, &[], options)
                .unwrap_or_else(|error| panic!("healthy bounded HNSW prepared query: {error:?}"))
                .result,
            Route::GovernedPrepared => engine
                .query_prepared_governed_fallible_view(
                    source, prepared, &[], options, &QueryGovernors::METERED,
                )
                .unwrap_or_else(|error| panic!("healthy bounded HNSW governed query: {error:?}"))
                .result,
        }
    }

    #[test]
    fn native_rank_membership_absence_and_post_rank_limit_are_bounded_on_host_routes() {
        let (space, terms) = space(1024);
        let TermValue::Iri(seed) = &terms[7] else { panic!("IRI fixture"); };
        let registry = registry(&space);
        let full_query = format!(
            "SELECT ?neighbour ?distance WHERE {{ ?neighbour <{NEAREST}> (<{seed}> 8 ?distance) }}"
        );
        let full = answer(&registry, &full_query);
        assert_eq!(full.len(), 8);
        let Some(TermValue::Iri(candidate)) = &full[4][0] else { panic!("IRI candidate"); };
        let queries = [
            full_query.clone(),
            format!(
                "SELECT ?distance WHERE {{ <{candidate}> <{NEAREST}> (<{seed}> ?k ?distance) }}"
            ),
            format!(
                "SELECT ?neighbour WHERE {{ ?neighbour <{NEAREST}> \
                 (<https://example.org/missing-seed> 8 ?distance) }}"
            ),
            format!(
                "SELECT ?distance WHERE {{ <{candidate}> <{NEAREST}> (<{seed}> 8 ?distance) }} LIMIT 1"
            ),
        ];
        let image = image();
        for (case, query) in queries.iter().enumerate() {
            let expected = answer(&registry, query);
            if case == 2 { assert!(expected.is_empty()); }
            if case == 3 {
                assert_eq!(expected, vec![vec![full[4][1].clone()]]);
            }
            let engine = NativeSparqlEngine::new();
            let env = ExtensionEnv::over_relations(registry.clone()).unwrap();
            let options = QueryOptions::new().with_env(&env);
            let prepared = engine.prepare_query_with_options(query, None, options).unwrap();
            for route in ROUTES {
                let window = CurrentThreadWindow::open();
                let source = open(&image, CEILING);
                let result = run(&engine, &source, query, &prepared, options, route);
                let measured = window.close();
                let peak = source.evidence().peak_bytes();
                assert!(
                    u64::try_from(measured.peak_working_bytes).unwrap() <= peak,
                    "case {case}: native allocations exceeded admitted peak"
                );
                assert!(peak <= CEILING);
                assert_eq!(result.solutions().unwrap().1, expected.as_slice());
            }
        }
    }

    #[test]
    fn cloned_and_extracted_native_answers_retain_without_copying_payloads() {
        let (space, terms) = space(64);
        let TermValue::Iri(seed) = &terms[3] else { panic!("IRI fixture"); };
        let registry = registry(&space);
        let query = format!(
            "SELECT ?neighbour ?distance WHERE {{ ?neighbour <{NEAREST}> (<{seed}> 6 ?distance) }}"
        );
        let expected = answer(&registry, &query);
        let image = image();
        let source = open(&image, CEILING);
        let engine = NativeSparqlEngine::new();
        let env = ExtensionEnv::over_relations(registry.clone()).unwrap();
        let options = QueryOptions::new().with_env(&env);
        let prepared = engine.prepare_query_with_options(&query, None, options).unwrap();
        let result = run(&engine, &source, &query, &prepared, options, Route::Prepared);
        let window = CurrentThreadWindow::open();
        let clone = result.clone();
        let retained = clone.into_solutions().unwrap();
        let measured = window.close();
        assert_eq!(measured.allocations, 0);
        assert_eq!(measured.requested_bytes, 0);
        assert_eq!(measured.retained_bytes, 0);
        drop(result);
        drop(prepared);
        drop(engine);
        drop(env);
        drop(registry);
        drop(space);
        drop(source);
        drop(image);
        assert_eq!(retained.as_parts().1, expected.as_slice());
    }

    fn observed_registry(
        space: &Arc<HnswSpace>,
    ) -> (PropertyFunctionRegistry, Arc<HnswObservations>) {
        let relation = space.relation();
        let observations = relation.observations();
        let mut registry = PropertyFunctionRegistry::new();
        registry.register(NEAREST.to_owned(), Arc::new(relation));
        (registry, observations)
    }

    #[test]
    fn first_native_traversal_admission_refusal_is_typed_and_releases_scratch() {
        let (space, terms) = space(4096);
        let TermValue::Iri(seed) = &terms[7] else { panic!("IRI fixture"); };
        let (registry, observations) = observed_registry(&space);
        let query = format!(
            "SELECT ?neighbour ?distance WHERE {{ ?neighbour <{NEAREST}> (<{seed}> 8 ?distance) }}"
        );
        let image = image();
        let engine = NativeSparqlEngine::new();
        let env = ExtensionEnv::over_relations(registry).unwrap();
        let options = QueryOptions::new().with_env(&env);
        let prepared = engine.prepare_query_with_options(&query, None, options).unwrap();
        let healthy = open(&image, CEILING);
        let opening = healthy.evidence().live_bytes();
        let before = observations.searches();
        let result = run(&engine, &healthy, &query, &prepared, options, Route::Prepared);
        assert_eq!(observations.searches(), before + 1);
        assert_eq!(result.solutions().unwrap().1.len(), 8);
        let successful_peak = healthy.evidence().peak_bytes();
        drop(result);
        drop(healthy);
        assert!(successful_peak > opening);

        // Find the lowest physical capacity that actually enters the beam.
        // Every probe is a cold source with the identical immutable prepared plan.
        let mut low = opening;
        let mut high = successful_peak;
        while low < high {
            let ceiling = low + (high - low) / 2;
            let source = open(&image, ceiling);
            let before = observations.searches();
            let outcome = engine.query_prepared_fallible_view(&source, &prepared, &[], options);
            let entered = observations.searches() != before;
            match &outcome {
                Ok(_) => {}
                Err(FallibleSparqlError::Operational {
                    error: SegmentedError::Residency { requested, limit }, ..
                }) => {
                    assert!(*requested > *limit);
                    assert_eq!(*limit, ceiling);
                }
                other => panic!("capacity refusal lost its physical cause: {other:?}"),
            }
            drop(outcome);
            if entered { high = ceiling; } else { low = ceiling + 1; }
        }

        let window = CurrentThreadWindow::open();
        let source = open(&image, low);
        let opening_live = source.evidence().live_bytes();
        let before_searches = observations.searches();
        let before_candidates = observations.graph_candidates();
        let error = engine.query_prepared_fallible_view(&source, &prepared, &[], options).unwrap_err();
        let measured = window.close();
        assert_eq!(observations.searches(), before_searches + 1);
        assert_eq!(
            observations.graph_candidates(), before_candidates,
            "the traversal refused before publishing successful candidate work"
        );
        assert!(matches!(
            &error,
            FallibleSparqlError::Operational {
                error: SegmentedError::Residency { requested, limit }, ..
            } if requested > limit && *limit == low
        ));
        assert!(u64::try_from(measured.peak_working_bytes).unwrap() <= source.evidence().peak_bytes());
        assert!(source.evidence().peak_bytes() <= low);
        drop(error);
        assert_eq!(
            source.evidence().live_bytes(), opening_live,
            "a failed native beam must release every query-owned grant"
        );
    }
}
