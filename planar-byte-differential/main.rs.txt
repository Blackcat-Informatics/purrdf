mod trace {
    use std::cell::RefCell;
    thread_local! {
        static INPUTS: RefCell<Vec<Vec<u8>>> = const { RefCell::new(Vec::new()) };
    }
    pub fn fold(state: u64, bytes: &[u8]) -> u64 {
        INPUTS.with_borrow_mut(|inputs| inputs.push(bytes.to_vec()));
        purrdf_hash::fnv::fold(state, bytes)
    }
    pub fn collect(digest: impl FnOnce() -> u64) -> (u64, Vec<Vec<u8>>) {
        INPUTS.with_borrow_mut(Vec::clear);
        let value = digest();
        (value, INPUTS.with_borrow_mut(core::mem::take))
    }
}

fn category(index: usize) -> &'static str {
    match index {
        0..20 => "WKT",
        20..40 => "GeoJSON",
        40..440 => "DE9IM",
        440..500 => "exact measure",
        500..900 => "distance",
        _ => "constructor or binary64",
    }
}

fn main() {
    assert_eq!(original_digest::corpus_len(), 20);
    assert_eq!(native_digest::corpus_len(), original_digest::corpus_len());
    let (original, expected) = trace::collect(original_digest::digest);
    assert_eq!(original, 0x9667_c2ee_2cd3_ad4b, "unchanged oracle must retain its accepted digest");
    let (native, actual) = trace::collect(native_digest::digest);
    println!("original={original:016x} native={native:016x} original_inputs={} native_inputs={}", expected.len(), actual.len());
    assert_eq!(expected.len(), actual.len(), "all original producer calls must remain present");
    let mut differences = 0;
    for (i, (old, new)) in expected.iter().zip(&actual).enumerate() {
        if old != new {
            differences += 1;
            println!("input={i} category={} original={old:?} native={new:?}", category(i));
        }
    }
    println!("differing_inputs={differences}");
    assert_eq!(differences, 0, "native output must preserve every original folded byte input");
}
