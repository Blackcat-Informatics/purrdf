    #[test]
    fn ground_and_paired_clone_join_enclosing_original_memory_without_owner_array() {
        use purrdf_lex::allocation::Memory;
        use purrdf_sparql_algebra::GroundTriple;
        let input = native_grounding_input(8);
        let (workspace, ledger, base) = account(1_000_000);
        let capability = workspace.capability();
        let mut frame = super::LexicalFrame::new(&capability);
        let (text, ground, clone) = {
            let mut memory = Memory::new(&mut frame);
            let text = memory.string("original authored query payload").unwrap();
            let initial = memory.live_bytes();
            let ground = crate::substitute::ground_term_from_value_with_memory(&input, &mut memory).unwrap();
            assert_eq!(memory.live_bytes(), initial + 8 * size_of::<GroundTriple>());
            let clone = crate::substitute::clone_ground_with_memory(&ground, &mut memory).unwrap();
            assert_eq!(memory.live_bytes(), initial + 16 * size_of::<GroundTriple>());
            (text, ground, clone)
        };
        let original = (text, ground, clone, frame.into_allocation());
        assert!(ledger.live.load(Ordering::Relaxed) > base);
        drop(capability);
        drop(workspace);
        assert!(ledger.live.load(Ordering::Relaxed) > 0);
        drop(original);
        assert_eq!(ledger.live.load(Ordering::Relaxed), 0);
    }

    #[test]
    fn grounding_refusal_destroys_partial_children_and_moves_first_cause_once() {
        let input = native_grounding_input(128);
        let base = QueryWorkspace::<Refusal>::control_bytes().unwrap();
        for extra in [0, 1, 128, 1024, 4096] {
            let (workspace, ledger, _) = account(base + extra);
            let capability = workspace.capability();
            assert!(matches!(
                crate::substitute::ground_term_from_value_admitted(&input, &capability),
                Err(crate::substitute::GroundFailure::Operational(EvalError::WorkspaceStopped))
            ));
            assert_eq!(ledger.live.load(Ordering::Relaxed), base);
            let cause = workspace.take_failure().expect("original admission cause");
            assert_eq!(cause.identity, 7);
            assert_eq!(ledger.clones.load(Ordering::Relaxed), 0);
            assert!(workspace.take_failure().is_none());
            drop(capability);
            drop(workspace);
            assert_eq!(ledger.live.load(Ordering::Relaxed), 0);
        }
    }

    /// Deliberately non-latching backing failure: the engine must retain the
    /// original typed cause without cloning or formatting it.
    struct GroundFaultView<'a> {
        inner: &'a purrdf_core::RdfDataset,
        failed_id: TermId,
        clones: Arc<AtomicUsize>,
    }

    impl purrdf_core::DatasetView for GroundFaultView<'_> {
        type Id = TermId;
        type ReadError = Refusal;
        type TermGuard<'a> = purrdf_core::TermRef<'a> where Self: 'a;
        type ProbePlan = purrdf_core::QuadProbePlan;

        fn quads(&self) -> impl Iterator<Item = purrdf_core::QuadIds> + '_ {
            purrdf_core::DatasetView::quads(self.inner)
        }
        fn resolve(&self, id: TermId) -> Result<Self::TermGuard<'_>, Refusal> {
            if id == self.failed_id {
                Err(Refusal { identity: 41, clones: self.clones.clone() })
            } else {
                Ok(self.inner.resolve(id))
            }
        }
        fn term_id_by_value(&self, value: &purrdf_core::TermValue) -> Result<Option<TermId>, Refusal> {
            Ok(self.inner.term_id_by_value(value))
        }
        fn capabilities(&self) -> purrdf_core::RdfStoreCapabilities {
            purrdf_core::DatasetView::capabilities(self.inner)
        }
        fn probe_plan(&self, s: bool, p: bool, o: bool, g: purrdf_core::GraphMatch) -> Self::ProbePlan {
            purrdf_core::DatasetView::probe_plan(self.inner, s, p, o, g)
        }
        fn quads_for_pattern_with_plan(
            &self, plan: &Self::ProbePlan, s: Option<TermId>, p: Option<TermId>,
            o: Option<TermId>, g: purrdf_core::GraphMatch,
        ) -> impl Iterator<Item = purrdf_core::QuadIds> + '_ {
            purrdf_core::DatasetView::quads_for_pattern_with_plan(self.inner, plan, s, p, o, g)
        }
        fn term_count(&self) -> u64 { purrdf_core::DatasetView::term_count(self.inner) }
    }

    #[test]
    fn borrowed_id_grounding_moves_exact_source_cause_and_releases_partial_triple() {
        use purrdf_core::{RdfDatasetBuilder, RdfLiteral};
        let mut builder = RdfDatasetBuilder::new();
        let subject = builder.intern_iri("http://example.org/s");
        let predicate = builder.intern_iri("http://example.org/p");
        let object = builder.intern_iri("http://example.org/o");
        let completed_subject = builder.intern_triple(subject, predicate, object);
        let datatype = builder.intern_iri(purrdf_xsd::datatype::XSD_STRING);
        let literal = builder.intern_literal(RdfLiteral::simple("literal"));
        let root = builder.intern_triple(completed_subject, predicate, literal);
        builder.push_quad(subject, predicate, root, None);
        let dataset = builder.freeze().expect("a valid shallow dataset");
        let (workspace, ledger, base) = account(1_000_000);
        let capability = workspace.capability();
        let view = GroundFaultView {
            inner: dataset.as_ref(), failed_id: datatype, clones: ledger.clones.clone(),
        };
        let failure = crate::substitute::ground_term_from_id_admitted(
            &view, root, &capability,
            |cause| crate::substitute::GroundFailure::Operational(workspace.source_error(cause)),
        );
        assert!(matches!(failure, Err(crate::substitute::GroundFailure::Operational(EvalError::WorkspaceStopped))));
        assert_eq!(ledger.live.load(Ordering::Relaxed), base);
        let cause = workspace.take_failure().expect("original source cause moves once");
        assert_eq!(cause.identity, 41);
        assert_eq!(ledger.clones.load(Ordering::Relaxed), 0);
        assert!(workspace.take_failure().is_none());
        drop(capability);
        drop(workspace);
        assert_eq!(ledger.live.load(Ordering::Relaxed), 0);
    }
