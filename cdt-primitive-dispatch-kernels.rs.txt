fn get_with_memory(value: &CdtValue, argument: &CdtTerm, memory: &mut Memory<'_>) -> Result<CdtOutcome<CdtTerm>, StorageError> {
    match value.contents() {
        CdtContents::List(items) => list_get_with_memory(items, argument, memory),
        CdtContents::Map(entries) => map_get_with_memory(entries, argument, memory),
    }
}
fn contains_with_memory(value: &CdtValue, term: &CdtTerm, memory: &mut Memory<'_>) -> Result<CdtOutcome<bool>, StorageError> {
    match value.contents() {
        CdtContents::List(items) => list_contains_with_memory(items, term, memory),
        CdtContents::Map(_) => Ok(raise("cdt:contains applies to a cdt:List; a cdt:Map has cdt:containsKey")),
    }
}

fn concat_with_memory(values: &[CdtValue], memory: &mut Memory<'_>) -> Result<CdtOutcome<CdtValue>, StorageError> {
    let mut lists: Vec<&[CdtTerm]> = Vec::new();
    memory.reserve(&mut lists, values.len())?;
    for value in values {
        match value.contents() {
            CdtContents::List(items) => lists.push(items),
            CdtContents::Map(_) => {
                memory.release_vec(lists)?;
                return Ok(raise("cdt:concat applies to cdt:List arguments only"));
            }
        }
    }
    let result = list_concat_with_memory(&lists, memory)?;
    memory.release_vec(lists)?;
    Ok(result)
}
