// Append to the existing native_regex_storage integration test, reusing its
// actual counting allocator and Account callback. Stage proposal, unrun.

fn compatibility(source: &str, flags: &str) -> xpath::OwnedCompatibilityPattern<Account> {
    xpath::compile_compatibility_with_storage(source, flags, Limits::without_presets(),
        Account::new(16_000_000)).unwrap()
}

fn compatibility_parity(source: &str, flags: &str, inputs: &[&str]) {
    let old = purrdf_core::xsd_regex::compile(source, flags).unwrap();
    let admitted = compatibility(source, flags);
    for &input in inputs {
        let found = admitted.value().find_with_storage(input, Limits::without_presets(),
            Account::new(16_000_000)).unwrap();
        let expected = old.as_regex().captures(input);
        assert_eq!(found.value().is_some(), expected.is_some(), "{source:?}, {flags:?}, {input:?}");
        if let (Some(actual), Some(expected)) = (found.value(), expected.as_ref()) {
            assert_eq!(actual.len(), expected.len());
            for capture in 0..actual.len() {
                assert_eq!(actual.get(capture), expected.get(capture).map(|span| span.range()),
                    "{source:?}, {flags:?}, {input:?}, capture {capture}");
            }
        }
        let matched = admitted.value().is_match_with_storage(input, Limits::without_presets(),
            Account::new(16_000_000)).unwrap();
        assert_eq!(*matched.value(), old.is_match(input), "{source:?}, {flags:?}, {input:?}");
    }
}

#[test]
fn compatibility_preserves_unicode16_simple_fold_properties_names_and_set_order() {
    let inputs = ["", "A", "a", "I", "i", "ı", "İ", "ſ", "s", "K", "K", "Σ", "σ", "ς",
        "é", ":", "_", "-", "\u{10000}", "\u{A7CE}", "\u{A7CF}"];
    for source in [r"\p{Lu}", r"\P{Lu}", r"\p{Ll}", r"\p{Cn}", r"\p{C}",
        r"\w", r"\W", r"\i", r"\I", r"\c", r"\C", r"[\i-[A-Z]]", r"[^A-Z-[IO]]",
        r"[a-z-[aeiou]]", r"\p{IsBasicLatin}", r"\P{IsBasicLatin}",
        r"\p{IsHighSurrogates}", r"\P{IsHighSurrogates}", "i", "s", "Σ", "\u{A7CE}"] {
        for flags in ["", "i"] { compatibility_parity(source, flags, &inputs); }
    }
    // These two letters were added in Unicode 17; compatibility remains at 16.
    let old = purrdf_core::xsd_regex::compile(r"\p{Lu}", "").unwrap();
    assert!(!old.is_match("\u{A7CE}"));
    assert!(*compatibility(r"\p{Cn}", "").value().is_match_with_storage("\u{A7CE}",
        Limits::without_presets(), Account::new(16_000_000)).unwrap().value());
    assert!(!*compatibility("\u{A7CE}", "i").value().is_match_with_storage("\u{A7CF}",
        Limits::without_presets(), Account::new(16_000_000)).unwrap().value());
}

#[test]
fn compatibility_preserves_final_newline_anchors_dot_q_x_and_liberal_escapes() {
    let inputs = ["", "a", "a\n", "\n", "a\r", "\r\n", "a b#c", "A B#C", "a.c", "aé", "&~"];
    for source in ["^", "$", "^$", "a$", "^a", ".", ".*", "a|$", "(?:a)?", r"\&\~", r"[a-\~]"] {
        for flags in ["", "i", "m", "s", "ims"] { compatibility_parity(source, flags, &inputs); }
    }
    for (source, flags) in [("a b#c", "qsimx"), ("a.c", "q"), ("I", "qi"), (" a  b # c ", "x")] {
        compatibility_parity(source, flags, &inputs);
    }
}

#[test]
fn compatibility_preserves_nullable_repeat_capture_priority() {
    let inputs = ["", "a", "aa", "aaa", "aaaa", "ab", "aab", "b", "é", "a\n"];
    for source in ["(a*)*", "(a?)*", "(a?)+", "((a?)*)", "(a*)+", "(a?){0,2}",
        "(a?){2,5}", "(a?){2,5}?", "((a)?){1,3}", "(a|aa)*b", "(|a)*", "(|a)+",
        "^(a*?)*$", "((a|)*)?", "(a*)*?", "(?:((a?)*)|a)"] {
        compatibility_parity(source, "", &inputs);
    }
}

#[test]
fn compatibility_replacement_preserves_empty_match_suppression_utf8_and_template_law() {
    for (source, flags) in [("", ""), ("a*", ""), ("a*|b", ""), ("(a*)*", ""),
        ("(a?){2,5}", ""), ("^$", "m"), ("$", "m"), ("", "q"), (".", "q")] {
        let old = purrdf_core::xsd_regex::compile(source, flags).unwrap();
        let admitted = compatibility(source, flags);
        for input in ["", "a", "aba", "aé", "éé", "a\n"] {
            for replacement in ["|", "$0", "$1:$0", "$23", r"\$\\", "é", "$", r"\x"] {
                let expected = old.replace_all(input, replacement);
                let actual = admitted.value().replace_all_with_storage(input, replacement,
                    Limits::without_presets(), Account::new(16_000_000), Account::new(16_000_000));
                match (expected, actual) {
                    (Ok(expected), Ok(actual)) => assert_eq!(actual.value().as_ref(), expected.as_ref(),
                        "{source:?}, {flags:?}, {input:?}, {replacement:?}"),
                    (Err(_), Err(actual)) => assert!(matches!(actual.value(), xpath::Error::Replacement(_))),
                    (expected, actual) => panic!("replacement parity: {source:?}, {flags:?}, {input:?}, {replacement:?}: {expected:?} / {actual:?}"),
                }
            }
        }
    }
}

#[test]
fn compatibility_refuses_original_non_grammar_constructs_and_keeps_dated_semantics() {
    for source in [r"(a)\1", r"\b", r"\x41", r"\0", "(?i)a", "(?=a)", "(?<name>a)",
        r"\p{IsGreek}", r"\p{Cs}", "[", "a{2,1}"] {
        assert!(purrdf_core::xsd_regex::compile(source, "").is_err(), "old {source}");
        let failure = xpath::compile_compatibility_with_storage(source, "", Limits::without_presets(),
            Account::new(16_000_000)).unwrap_err();
        assert!(!failure.value().is_operational(), "grammar {source}");
    }
    let old = xpath::compile(Profile::Xpath31, "(a)\\1", "", Limits::new()).unwrap();
    assert!(old.is_match("aa", Limits::new()).unwrap());
    assert!(xpath::compile(Profile::Xpath31, r"\&", "", Limits::new()).is_err());
    assert!(xpath::compile(Profile::Xpath31, "^$", "m", Limits::new()).unwrap()
        .find("a\n", Limits::new()).unwrap().is_none());
    assert!(matches!(xpath::compile(Profile::Xpath31, "", "", Limits::new()).unwrap()
        .replace_all("a", "x", Limits::new()), Err(xpath::Error::EmptyMatch)));
}

#[test]
fn compatibility_native_compiler_and_execution_cover_actual_physical_allocations() {
    let source = "((a?){2,5}[A-Z-[IO]]?)*\\p{Lu}\\i".repeat(8);
    let window = CurrentThreadWindow::open();
    let admitted = xpath::compile_compatibility_with_storage(&source, "ix", Limits::without_presets(),
        Account::new(16_000_000)).unwrap();
    let measured = window.close();
    assert!(usize::try_from(measured.peak_working_bytes).unwrap() <= admitted.storage().peak);
    assert_eq!(usize::try_from(measured.retained_bytes).unwrap(), admitted.storage().live);
    let compiled = compatibility("(a*)*", "");
    let window = CurrentThreadWindow::open();
    let captured = compiled.value().find_with_storage("aaaa", Limits::without_presets(),
        Account::new(16_000_000)).unwrap();
    let measured = window.close();
    assert!(usize::try_from(measured.peak_working_bytes).unwrap() <= captured.storage().peak);
    assert_eq!(usize::try_from(measured.retained_bytes).unwrap(), captured.storage().live);
    let window = CurrentThreadWindow::open();
    let replaced = compiled.value().replace_all_with_storage("aéaé", "$1|",
        Limits::without_presets(), Account::new(16_000_000), Account::new(16_000_000)).unwrap();
    let measured = window.close();
    assert!(usize::try_from(measured.peak_working_bytes).unwrap()
        <= replaced.storage().0.peak + replaced.storage().1.peak);
    assert_eq!(usize::try_from(measured.retained_bytes).unwrap(), replaced.storage().1.live);
}

#[test]
fn compatibility_physical_refusal_and_native_diagnostics_keep_original_accounts() {
    for source in ["(a*)*", "(?<an_unrecognized_group>", r"(a)\999999999999999999999999999"] {
        let window = CurrentThreadWindow::open();
        let failure = xpath::compile_compatibility_with_storage(source, "", Limits::without_presets(),
            Account::new(0)).unwrap_err();
        let measured = window.close();
        assert!(matches!(failure.value(), xpath::Error::Storage(StorageError::AdmissionFailed)));
        assert_eq!(measured.peak_working_bytes, 0);
        assert_eq!(measured.retained_bytes, 0);
        assert!(failure.storage().refused.is_some());
    }
    let pattern = compatibility("(a*)*", "");
    let window = CurrentThreadWindow::open();
    let failure = pattern.value().replace_all_with_storage("aé", "|", Limits::without_presets(),
        Account::new(0), Account::new(16_000_000)).unwrap_err();
    let measured = window.close();
    assert!(matches!(failure.value(), xpath::Error::Storage(StorageError::AdmissionFailed)));
    assert_eq!(measured.peak_working_bytes, 0);
    assert!(failure.storage().0.refused.is_some());
}
